    def accuracy_score(y_true, y_pred):

    def f1_score(y_true, y_pred, average='weighted', zero_division=0):
        y_true, y_pred = np.array(y_true), np.array(y_pred)
        unique_labels = np.unique(np.concatenate([y_true, y_pred]))
        
        f1_scores = []
        for label in unique_labels:
            tp = np.sum((y_true == label) & (y_pred == label))
            fp = np.sum((y_true != label) & (y_pred == label))
            fn = np.sum((y_true == label) & (y_pred != label))
            
            precision = tp / (tp + fp) if (tp + fp) > 0 else zero_division
            recall = tp / (tp + fn) if (tp + fn) > 0 else zero_division
            f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else zero_division
            f1_scores.append(f1)
        
        if average == 'weighted':
            weights = [np.sum(y_true == label) for label in unique_labels]
            return np.average(f1_scores, weights=weights)
        return np.mean(f1_scores)
    
    def confusion_matrix(y_true, y_pred):
        y_true, y_pred = np.array(y_true), np.array(y_pred)
        labels = np.unique(np.concatenate([y_true, y_pred]))
        n = len(labels)
        cm = np.zeros((n, n), dtype=int)
        for i, true_label in enumerate(labels):
            for j, pred_label in enumerate(labels):
                cm[i, j] = np.sum((y_true == true_label) & (y_pred == pred_label))
        return cm

# ==== 1. ENHANCED DATASET DISCOVERY SYSTEM ====
class DatasetExplorer:
    """Comprehensive dataset explorer to analyze ALL existing data."""
    
    def __init__(self, data_roots):
        self.data_roots = data_roots
        self.discovered_files = {}
        self.dataset_stats = {}
        self.image_validator = self._create_image_validator()
        
    def _create_image_validator(self):
        """Create robust image validator."""
        def validate_image(img_path):
            try:
                if not os.path.exists(img_path):
                    return False, "File not found"
                
                with Image.open(img_path) as img:
                    width, height = img.size
                    mode = img.mode
                    
                    if width < 32 or height < 32:
                        return False, "Too small"
                    if width > 4096 or height > 4096:
                        return False, "Too large"
                    
                    return True, {
                        'size': (width, height),
                        'mode': mode,
                        'format': img.format
                    }
            except Exception as e:
                return False, str(e)
        
        return validate_image
    
    def discover_all_datasets(self):
        """Discover ALL files in each dataset comprehensively."""
        print("🔍 COMPREHENSIVE DATASET DISCOVERY - Finding ALL files...")
        
        total_discovered = 0
        
        for dataset_name, dataset_path in self.data_roots.items():
            print(f"\n📂 Exploring {dataset_name.upper()}...")
            files_info = self._explore_dataset_structure(dataset_name, dataset_path)
            self.discovered_files[dataset_name] = files_info
            
            valid_count = sum(1 for f in files_info if f['is_valid'])
            total_discovered += len(files_info)
            
            print(f"   📊 Found: {len(files_info):,} total files")
            print(f"   ✅ Valid: {valid_count:,} images")
            print(f"   ❌ Issues: {len(files_info) - valid_count:,} files")
        
        print(f"\n🎉 DISCOVERY COMPLETE: {total_discovered:,} files across all datasets")
        return self.discovered_files
    
    def _explore_dataset_structure(self, dataset_name, dataset_path):
        """Explore complete structure of a dataset."""
        dataset_path = Path(dataset_path)
        if not dataset_path.exists():
            print(f"   ⚠️ Path not found: {dataset_path}")
            return []
        
        # Discover ALL image files recursively
        image_extensions = {'.jpg', '.jpeg', '.png', '.bmp', '.tiff', '.webp'}
        all_files = []
        
        print(f"   🔍 Scanning directory structure...")
        for ext in image_extensions:
            for file_path in dataset_path.rglob(f"*{ext}"):
                all_files.append(file_path)
            for file_path in dataset_path.rglob(f"*{ext.upper()}"):
                all_files.append(file_path)
        
        # Remove duplicates
        all_files = list(set(all_files))
        
        print(f"   📁 Discovered {len(all_files):,} image files")
        
        # Analyze each file with progress bar
        files_info = []
        for file_path in tqdm(all_files, desc=f"Analyzing {dataset_name}", leave=False):
            is_valid, info = self.image_validator(file_path)
            
            file_info = {
                'path': str(file_path),
                'relative_path': str(file_path.relative_to(dataset_path)),
                'name': file_path.name,
                'size_mb': file_path.stat().st_size / (1024*1024) if file_path.exists() else 0,
                'is_valid': is_valid,
                'validation_info': info if is_valid else info,
                'dataset': dataset_name
            }
            
            # Extract metadata from path structure
            path_parts = file_path.relative_to(dataset_path).parts
            file_info['path_depth'] = len(path_parts)
            file_info['subdirectory'] = path_parts[0] if len(path_parts) > 1 else 'root'
            
            # Dataset-specific analysis
            if dataset_name == 'iddaw':
                file_info.update(self._analyze_iddaw_file(file_path, dataset_path))
            elif dataset_name == 'ots':
                file_info.update(self._analyze_ots_file(file_path, dataset_path))
            elif dataset_name == 'rtts':
                file_info.update(self._analyze_rtts_file(file_path, dataset_path))
            elif dataset_name == 'new_idd':
                file_info.update(self._analyze_newidd_file(file_path, dataset_path))
            
            files_info.append(file_info)
        
        return files_info
    
    def _analyze_iddaw_file(self, file_path, root_path):
        """Analyze IDD-AW specific file structure."""
        rel_path = file_path.relative_to(root_path)
        parts = rel_path.parts
        
        weather_type = 'UNKNOWN'
        split_type = 'UNKNOWN'
        modality = 'UNKNOWN'
        
        # Extract weather condition from path
        for part in parts:
            if part.upper() in ['FOG', 'RAIN', 'SNOW', 'LOWLIGHT']:
                weather_type = part.upper()
            elif part in ['train', 'val']:
                split_type = part
            elif part in ['rgb', 'nir', 'gtSeg']:
                modality = part
        
        return {
            'weather_type': weather_type,
            'split_type': split_type,
            'modality': modality,
            'sequence': parts[3] if len(parts) > 3 else 'unknown'
        }
    
    def _analyze_ots_file(self, file_path, root_path):
        """Analyze OTS specific file structure."""
        rel_path = file_path.relative_to(root_path)
        parts = rel_path.parts
        
        weather_type = 'UNKNOWN'
        if 'clear' in parts:
            weather_type = 'CLEAR'
        elif 'hazy' in parts:
            weather_type = 'HAZY'
        
        return {
            'weather_type': weather_type,
            'split_type': 'train',
            'modality': 'rgb'
        }
    
    def _analyze_rtts_file(self, file_path, root_path):
        """Analyze RTTS specific file structure."""
        return {
            'weather_type': 'HAZY',
            'split_type': 'train', 
            'modality': 'rgb'
        }
    
    def _analyze_newidd_file(self, file_path, root_path):
        """Analyze New IDD specific file structure."""
        modality = 'mask' if 'Mask_' in file_path.name else 'rgb'
        
        return {
            'weather_type': 'CLEAR',
            'split_type': 'train',
            'modality': modality
        }

class DatasetVisualizationDashboard:
    """Create comprehensive visualizations of current dataset state."""
    
    def __init__(self, discovered_files):
        self.discovered_files = discovered_files
        self.stats = self._compute_comprehensive_stats()
    
    def _compute_comprehensive_stats(self):
        """Compute comprehensive statistics across all datasets."""
        stats = {
            'total_files': 0,
            'valid_images': 0,
            'invalid_files': 0,
            'by_dataset': {},
            'by_weather': defaultdict(int),
            'by_modality': defaultdict(int),
            'by_format': defaultdict(int),
            'size_distribution': [],
            'resolution_distribution': []
        }
        
        for dataset_name, files in self.discovered_files.items():
            dataset_stats = {
                'total': len(files),
                'valid': sum(1 for f in files if f['is_valid']),
                'invalid': sum(1 for f in files if not f['is_valid']),
                'weather_breakdown': defaultdict(int),
                'modality_breakdown': defaultdict(int),
                'avg_size_mb': 0
            }
            
            valid_sizes = []
            for file_info in files:
                stats['total_files'] += 1
                
                if file_info['is_valid']:
                    stats['valid_images'] += 1
                    valid_sizes.append(file_info['size_mb'])
                    
                    # Collect validation info for resolution analysis
                    if isinstance(file_info['validation_info'], dict):
                        stats['resolution_distribution'].append(file_info['validation_info']['size'])
                        
                        format_name = file_info['validation_info'].get('format', 'UNKNOWN')
                        stats['by_format'][format_name] += 1
                else:
                    stats['invalid_files'] += 1
                
                # Weather and modality stats
                weather = file_info.get('weather_type', 'UNKNOWN')
                modality = file_info.get('modality', 'UNKNOWN')
                
                stats['by_weather'][weather] += 1
                stats['by_modality'][modality] += 1
                
                dataset_stats['weather_breakdown'][weather] += 1
                dataset_stats['modality_breakdown'][modality] += 1
            
            if valid_sizes:
                dataset_stats['avg_size_mb'] = np.mean(valid_sizes)
                stats['size_distribution'].extend(valid_sizes)
            
            stats['by_dataset'][dataset_name] = dataset_stats
        
        return stats
    
    def create_comprehensive_dashboard(self, save_path=None):
        """Create comprehensive current dataset analysis dashboard."""
        fig = plt.figure(figsize=(24, 20))
        gs = fig.add_gridspec(4, 4, hspace=0.35, wspace=0.3)
        fig.suptitle('📊 COMPREHENSIVE CURRENT DATASET ANALYSIS (BEFORE CELL 3)', 
                    fontsize=20, fontweight='bold', y=0.98)
        
        # Row 1: Dataset Overview
        self._plot_dataset_overview(fig, gs)
        
        # Row 2: Weather & Modality Distribution
        self._plot_weather_modality_distribution(fig, gs)
        
        # Row 3: Technical Analysis
        self._plot_technical_analysis(fig, gs)
        
        # Row 4: Problem Analysis (IDD-AW Investigation)
        self._plot_problem_analysis(fig, gs)
        
        if save_path:
            plt.savefig(save_path, dpi=200, bbox_inches='tight', facecolor='white')
            print(f"📊 Comprehensive dashboard saved: {save_path}")
        
        plt.show()
    
    def _plot_dataset_overview(self, fig, gs):
        """Plot dataset overview section."""
        # Dataset Size Comparison (Pie Chart)
        ax1 = fig.add_subplot(gs[0, 0])
        dataset_sizes = [self.stats['by_dataset'][ds]['total'] for ds in self.stats['by_dataset']]
        dataset_names = list(self.stats['by_dataset'].keys())
        
        colors = ['#FF6B6B', '#4ECDC4', '#45B7D1', '#96CEB4']
        wedges, texts, autotexts = ax1.pie(dataset_sizes, labels=dataset_names, autopct='%1.1f%%', 
                                          colors=colors, startangle=90)
        ax1.set_title('Dataset Size Distribution\n(Total Files Found)', fontweight='bold')
        
        # Valid vs Invalid Files
        ax2 = fig.add_subplot(gs[0, 1])
        valid_invalid = [self.stats['valid_images'], self.stats['invalid_files']]
        labels = ['Valid Images', 'Invalid/Corrupt Files']
        colors = ['lightgreen', 'lightcoral']
        
        bars = ax2.bar(labels, valid_invalid, color=colors, alpha=0.8)
        ax2.set_title('Image Validation Status', fontweight='bold')
        ax2.set_ylabel('Number of Files')
        
        # Add value labels
        for bar, value in zip(bars, valid_invalid):
            height = bar.get_height()
            ax2.text(bar.get_x() + bar.get_width()/2., height + max(valid_invalid)*0.01,
                    f'{value:,}', ha='center', va='bottom', fontweight='bold')
        
        # Dataset Breakdown Table
        ax3 = fig.add_subplot(gs[0, 2:])
        ax3.axis('off')
        
        table_data = []
        headers = ['Dataset', 'Total Files', 'Valid Images', 'Invalid Files', 'Success Rate', 'Avg Size (MB)']
        
        for dataset_name, stats in self.stats['by_dataset'].items():
            success_rate = (stats['valid'] / stats['total'] * 100) if stats['total'] > 0 else 0
            table_data.append([
                dataset_name.upper(),
                f"{stats['total']:,}",
                f"{stats['valid']:,}",
                f"{stats['invalid']:,}",
                f"{success_rate:.1f}%",
                f"{stats['avg_size_mb']:.2f}"
            ])
        
        table = ax3.table(cellText=table_data, colLabels=headers, loc='center', cellLoc='center')
        table.auto_set_font_size(False)
        table.set_fontsize(9)
        table.scale(1.2, 1.8)
        
        # Style the table
        for i in range(len(headers)):
            table[(0, i)].set_facecolor('#4472C4')
            table[(0, i)].set_text_props(weight='bold', color='white')
        
        ax3.set_title('Detailed Dataset Breakdown', fontweight='bold', pad=20)
    
    def _plot_weather_modality_distribution(self, fig, gs):
        """Plot weather and modality distributions."""

        ax1 = fig.add_subplot(gs[1, 0:2])
        weather_types = list(self.stats['by_weather'].keys())
        weather_counts = list(self.stats['by_weather'].values())
        
        bars = ax1.bar(weather_types, weather_counts, color='skyblue', alpha=0.8, edgecolor='navy')
        ax1.set_title('Weather Condition Distribution (Current Data)', fontweight='bold')
        ax1.set_ylabel('Number of Files')
        ax1.tick_params(axis='x', rotation=45)
        
        # Add value labels
        for bar, value in zip(bars, weather_counts):
            height = bar.get_height()
            ax1.text(bar.get_x() + bar.get_width()/2., height + max(weather_counts)*0.01,
                    f'{value:,}', ha='center', va='bottom', fontweight='bold')
        
        # Modality Distribution
        ax2 = fig.add_subplot(gs[1, 2:])
        modality_types = list(self.stats['by_modality'].keys())
        modality_counts = list(self.stats['by_modality'].values())
        
        bars = ax2.bar(modality_types, modality_counts, color='lightcoral', alpha=0.8, edgecolor='darkred')
        ax2.set_title('Data Modality Distribution (RGB/NIR/Masks)', fontweight='bold')
        ax2.set_ylabel('Number of Files')
        ax2.tick_params(axis='x', rotation=45)
        
        # Add value labels
        for bar, value in zip(bars, modality_counts):
            height = bar.get_height()
            ax2.text(bar.get_x() + bar.get_width()/2., height + max(modality_counts)*0.01,

    def _plot_technical_analysis(self, fig, gs):
        """Plot technical analysis of current data."""
        # File Size Distribution
        ax1 = fig.add_subplot(gs[2, 0:2])
        if self.stats['size_distribution']:
            ax1.hist(self.stats['size_distribution'], bins=50, alpha=0.7, color='green', edgecolor='black')
            ax1.set_title('File Size Distribution (MB)', fontweight='bold')
            ax1.set_xlabel('File Size (MB)')
            ax1.set_ylabel('Frequency')
            ax1.axvline(np.mean(self.stats['size_distribution']), color='red', linestyle='--', 
                       label=f'Mean: {np.mean(self.stats["size_distribution"]):.2f}MB')
            ax1.legend()
        else:
            ax1.text(0.5, 0.5, 'No size data available', ha='center', va='center', 
                    transform=ax1.transAxes, fontsize=12)
            ax1.set_title('File Size Distribution', fontweight='bold')
        
        # Resolution Analysis
        ax2 = fig.add_subplot(gs[2, 2:])
        if self.stats['resolution_distribution']:
            widths = [res[0] for res in self.stats['resolution_distribution']]
            heights = [res[1] for res in self.stats['resolution_distribution']]
            
            scatter = ax2.scatter(widths, heights, alpha=0.6, s=20, c='purple')
            ax2.set_title('Image Resolution Distribution', fontweight='bold')
            ax2.set_xlabel('Width (pixels)')
            ax2.set_ylabel('Height (pixels)')
            
            # Add common resolutions as reference lines
            common_resolutions = [(512, 512), (1024, 1024), (1920, 1080), (640, 480)]
            for w, h in common_resolutions:
                if any(abs(width-w) < 100 and abs(height-h) < 100 for width, height in zip(widths, heights)):
                    ax2.axhline(h, color='red', alpha=0.3, linestyle='--')
                    ax2.axvline(w, color='red', alpha=0.3, linestyle='--')
                    ax2.text(w, h, f'{w}x{h}', fontsize=8, ha='left', va='bottom', color='red')
        else:
            ax2.text(0.5, 0.5, 'No resolution data available', ha='center', va='center', 
                    transform=ax2.transAxes, fontsize=12)
            ax2.set_title('Image Resolution Distribution', fontweight='bold')

# ==== 3. SAMPLE GRID VISUALIZATION ====
def create_current_dataset_sample_grid(discovered_files, samples_per_dataset=4, save_path=None):
    """Create sample grid showing current dataset images (before Cell 3 processing)."""
    print("🎨 Creating current dataset sample grid...")
    
    datasets_with_samples = {}
    
    # Collect sample images from each dataset
    for dataset_name, files in discovered_files.items():
        valid_files = [f for f in files if f['is_valid'] and f.get('modality', 'rgb') == 'rgb']
        
        if len(valid_files) >= samples_per_dataset:
            # Sample from different weather conditions if available
            samples_by_weather = defaultdict(list)
            for file_info in valid_files:
                weather = file_info.get('weather_type', 'UNKNOWN')
                samples_by_weather[weather].append(file_info)
            
            selected_samples = []
            for weather, files_list in samples_by_weather.items():
                if len(selected_samples) < samples_per_dataset:
                    selected_samples.extend(files_list[:min(2, len(files_list))])
            
            # Fill remaining slots randomly
            while len(selected_samples) < samples_per_dataset:
                remaining = [f for f in valid_files if f not in selected_samples]
                if remaining:
                    selected_samples.append(random.choice(remaining))
                else:
                    break
            
            datasets_with_samples[dataset_name] = selected_samples[:samples_per_dataset]
    
    if not datasets_with_samples:
        print("⚠️ No valid samples found for visualization")
        return False
    
    # Create visualization
    num_datasets = len(datasets_with_samples)
    fig, axes = plt.subplots(num_datasets, samples_per_dataset, figsize=(20, 5*num_datasets))
    if num_datasets == 1:
        axes = axes.reshape(1, -1)
    elif samples_per_dataset == 1:
        axes = axes.reshape(-1, 1)
    
    fig.suptitle('🖼️ CURRENT DATASET SAMPLES (All Existing Data Before Cell 3)', 
                fontsize=16, fontweight='bold')
    
    for row, (dataset_name, samples) in enumerate(datasets_with_samples.items()):
        for col, sample in enumerate(samples):
            try:
                # Load image
                img_path = sample['path']
                with Image.open(img_path) as img:
                    img_rgb = img.convert('RGB')
                    img_array = np.array(img_rgb)
                
                axes[row, col].imshow(img_array)
                
                # Create informative title
                weather = sample.get('weather_type', 'UNK')
                size_info = sample['validation_info']['size'] if isinstance(sample['validation_info'], dict) else 'UNK'
                size_mb = sample['size_mb']
                
                title = f"{dataset_name.upper()}\n{weather}\n{size_info} | {size_mb:.1f}MB"
                axes[row, col].set_title(title, fontsize=9)
                axes[row, col].axis('off')
                
            except Exception as e:
                axes[row, col].text(0.5, 0.5, f'Load Error\n{dataset_name}', 
                                   ha='center', va='center', transform=axes[row, col].transAxes,
                                   fontsize=10, color='red')
                axes[row, col].set_title(f'{dataset_name.upper()}\nERROR', fontsize=9, color='red')
                axes[row, col].axis('off')
    
    plt.tight_layout()
    
    if save_path:
        plt.savefig(save_path, dpi=200, bbox_inches='tight', facecolor='white')
        print(f"🖼️ Sample grid saved: {save_path}")
    
    plt.show()
    return True

# ==== 1. ADVANCED SYNTHETIC WEATHER GENERATION SYSTEM ====
class AdvancedWeatherSynthesizer:
    """State-of-the-art synthetic weather generation using advanced computer vision techniques."""
    
    def __init__(self):
        self.weather_effects = {
            'heavy_fog': self._create_heavy_fog,
            'light_fog': self._create_light_fog,
            'dense_fog': self._create_dense_fog,
            'light_rain': self._create_light_rain,
            'heavy_rain': self._create_heavy_rain,
            'storm_rain': self._create_storm_rain,
            'light_snow': self._create_light_snow,
            'heavy_snow': self._create_heavy_snow,
            'blizzard': self._create_blizzard,
            'light_haze': self._create_light_haze,
            'dense_haze': self._create_dense_haze,
            'smog': self._create_smog,
            'twilight': self._create_twilight,
            'dawn': self._create_dawn,
            'night_rain': self._create_night_rain,
            'winter_storm': self._create_winter_storm
        }
        
        # Weather type to label mapping
        self.weather_to_label = {
            'heavy_fog': 'FOG', 'light_fog': 'FOG', 'dense_fog': 'FOG',
            'light_rain': 'RAIN', 'heavy_rain': 'RAIN', 'storm_rain': 'RAIN', 'night_rain': 'RAIN',
            'light_snow': 'SNOW', 'heavy_snow': 'SNOW', 'blizzard': 'SNOW', 'winter_storm': 'SNOW',
            'light_haze': 'HAZE', 'dense_haze': 'HAZE', 'smog': 'HAZE',
            'twilight': 'LOWLIGHT', 'dawn': 'LOWLIGHT'
        }
        
        print(f"🎨 Advanced Weather Synthesizer initialized with {len(self.weather_effects)} weather types")
    
    def _create_heavy_fog(self, image):
        """Create heavy fog effect with realistic atmospheric scattering."""
        img_array = np.array(image, dtype=np.float32) / 255.0
        
        # Create fog density map (distance-based)
        h, w = img_array.shape[:2]
        y, x = np.ogrid[:h, :w]
        
        # Distance from bottom (road level)
        distance_from_bottom = (h - y) / h
        fog_density = 0.7 + 0.3 * distance_from_bottom
        
        # Add random variations
        noise = np.random.normal(0, 0.1, (h, w))
        fog_density = np.clip(fog_density + noise, 0.3, 0.95)
        
        # Apply fog effect
        fog_color = np.array([0.9, 0.9, 0.85])  # Slightly warm white
        for c in range(3):
            img_array[:, :, c] = img_array[:, :, c] * (1 - fog_density) + fog_color[c] * fog_density
        
        # Reduce contrast and saturation
        img_array = img_array * 0.8 + 0.1
        
        return (np.clip(img_array, 0, 1) * 255).astype(np.uint8)
    
    def _create_light_fog(self, image):
        """Create light fog with subtle atmospheric effects."""
        img_array = np.array(image, dtype=np.float32) / 255.0
        
        # Lighter fog density
        h, w = img_array.shape[:2]
        fog_density = 0.2 + 0.1 * np.random.random((h, w))

    def _create_heavy_rain(self, image):
        """Create heavy rain with realistic droplet effects and atmospheric darkening."""
        img_array = np.array(image, dtype=np.float32)

        img_array = img_array * 0.6 + 10

            thickness = np.random.randint(1, 3)

            for i in range(length):

        haze_intensity = 0.3

        for c in range(3):

    def _create_heavy_snow(self, image):
        """Create heavy snow with realistic snowflake patterns."""
        img_array = np.array(image, dtype=np.float32)
        
        # Brighten image for snow reflection
        img_array = img_array * 0.9 + 30
        
        h, w = img_array.shape[:2]
        
        # Create snowflakes of various sizes
        for _ in range(1500):
            x = np.random.randint(0, w)
            y = np.random.randint(0, h)
            size = np.random.randint(2, 8)
            intensity = 200 + np.random.randint(0, 55)
            
            # Draw snowflake
            for dx in range(-size//2, size//2 + 1):
                for dy in range(-size//2, size//2 + 1):
                    if (0 <= x + dx < w and 0 <= y + dy < h and 
                        dx*dx + dy*dy <= (size//2)**2):
                        alpha = max(0, 1 - (dx*dx + dy*dy) / (size*size/4))
                        img_array[y + dy, x + dx] = (
                            img_array[y + dy, x + dx] * (1 - alpha) + 
                            intensity * alpha
                        )
        
        # Add atmospheric snow haze
        haze = 0.2
        for c in range(3):
            img_array[:, :, c] = img_array[:, :, c] * (1 - haze) + 240 * haze
        
        return np.clip(img_array, 0, 255).astype(np.uint8)

    def _create_dense_haze(self, image):
        """Create dense haze with reduced visibility."""
        img_array = np.array(image, dtype=np.float32) / 255.0
        
        # Haze density varies with distance
        h, w = img_array.shape[:2]
        y, x = np.ogrid[:h, :w]
        
        # Distance-based haze (stronger at horizon)
        distance_factor = y / h
        haze_density = 0.4 + 0.4 * distance_factor
        
        # Add turbulence
        noise = np.random.normal(0, 0.05, (h, w))
        haze_density = np.clip(haze_density + noise, 0.2, 0.8)
        
        # Haze color (yellowish-brown)
        haze_color = np.array([0.85, 0.8, 0.65])
        
        for c in range(3):
            img_array[:, :, c] = img_array[:, :, c] * (1 - haze_density) + haze_color[c] * haze_density
        
        # Reduce contrast
        img_array = img_array * 0.7 + 0.15
        
        return (np.clip(img_array, 0, 1) * 255).astype(np.uint8)

    def _create_night_rain(self, image):

    def process_all_real_images(self):
        """Process ALL discovered real images into unified format."""
        all_real_metadata = []
        
        for dataset_name, files in self.discovered_files.items():
            print(f"\n📂 Processing {dataset_name.upper()}...")
            
            # Filter valid RGB images
            valid_rgb_files = [
                f for f in files 
                if f['is_valid'] and f.get('modality', 'rgb') == 'rgb'
            ]
            
            print(f"   📊 Found {len(valid_rgb_files):,} valid RGB images")
            
            # Process with progress bar
            for file_info in tqdm(valid_rgb_files, desc=f"Processing {dataset_name}", leave=False):
                try:
                    processed_metadata = self.standardize_image(file_info, dataset_name)
                    if processed_metadata:
                        all_real_metadata.append(processed_metadata)
                        self.processing_stats['real_images'] += 1
                        self.processing_stats['by_source_dataset'][dataset_name] += 1
                        
                        # Track weather distribution
                        weather_type = processed_metadata.get('weather_condition', 'UNKNOWN')
                        self.processing_stats['by_weather_type'][weather_type] += 1
                        
                except Exception as e:
                    print(f"⚠️ Failed to process {file_info['path']}: {e}")
                    self.processing_stats['failed_processing'] += 1
            
            print(f"   ✅ Processed: {self.processing_stats['by_source_dataset'][dataset_name]:,} images")
        
        self.processing_stats['total_processed'] = len(all_real_metadata)
        print(f"\n✅ REAL IMAGE PROCESSING COMPLETE: {len(all_real_metadata):,} images standardized")
        
        return all_real_metadata
    
    def standardize_image(self, file_info, source_dataset):
        """Convert any image to standardized 512x512 JPEG RGB format."""
        try:
            start_time = time.time()
            
            # Load and convert image
            with Image.open(file_info['path']) as img:
                # Convert to RGB
                if img.mode != 'RGB':
                    img = img.convert('RGB')
                
                # Resize with high-quality resampling
                img_resized = img.resize(self.target_size, Image.Resampling.LANCZOS)
                
                # Generate standardized filename
                img_id = f"{self.processing_stats['total_processed']:08d}"
                save_path = self.images_dir / f"{img_id}.jpg"
                
                # Save with high quality
                img_resized.save(save_path, 'JPEG', quality=95, optimize=True)
                
                # Calculate quality metrics
                file_size_mb = save_path.stat().st_size / (1024*1024)
                processing_time = time.time() - start_time
                
                # Store metrics
                self.processing_stats['processing_times'].append(processing_time)
                self.processing_stats['file_sizes_mb'].append(file_size_mb)
                
                # Create comprehensive metadata
                metadata = {
                    'image_id': img_id,
                    'filename': f"{img_id}.jpg",
                    'original_path': file_info['path'],
                    'source_dataset': source_dataset,
                    'original_size': img.size if 'img' in locals() else (0, 0),
                    'standardized_size': self.target_size,
                    'weather_condition': file_info.get('weather_type', 'UNKNOWN'),
                    'is_synthetic': False,
                    'modality': 'rgb',
                    'file_size_mb': file_size_mb,
                    'processing_time': processing_time,
                    'quality_score': self.calculate_quality_score(img_resized)
                }
                
                self.processing_stats['total_processed'] += 1
                return metadata
                
        except Exception as e:
            print(f"⚠️ Error standardizing {file_info['path']}: {e}")
            return None
    
    def calculate_quality_score(self, image):
        """Calculate simple quality score for image."""
        try:
            img_array = np.array(image)
            
            # Calculate metrics
            brightness = np.mean(img_array)
            contrast = np.std(img_array)
            
            # Simple quality score (0-100)
            quality_score = min(100, (brightness / 255 * 50) + (contrast / 128 * 50))
            return round(quality_score, 2)
            
        except:
            return 50.0  # Default score

        selected_base_images = random.sample(real_metadata, base_images_needed)
        
        generated_count = 0
        weather_effects = list(self.weather_synthesizer.weather_effects.keys())
        
        with tqdm(total=target_synthetic_count, desc="Generating synthetic weather") as pbar:

                        base_img = base_img.convert('RGB')
                        
                        # Generate multiple weather variations from this base
                        effects_per_base = min(3, target_synthetic_count - generated_count)
                        selected_effects = random.sample(weather_effects, min(effects_per_base, len(weather_effects)))
                        
                        for weather_effect in selected_effects:
                            if generated_count >= target_synthetic_count:
                                break
                            
                            try:
                                # Apply weather effect
                                weather_func = self.weather_synthesizer.weather_effects[weather_effect]
                                synthetic_img_array = weather_func(base_img)
                                synthetic_img = Image.fromarray(synthetic_img_array)
                                
                                # Save synthetic image
                                synthetic_id = f"{self.processing_stats['total_processed']:08d}"
                                save_path = self.images_dir / f"{synthetic_id}.jpg"
                                synthetic_img.save(save_path, 'JPEG', quality=95, optimize=True)
                                
                                # Create metadata
                                file_size_mb = save_path.stat().st_size / (1024*1024)
                                weather_label = self.weather_synthesizer.weather_to_label.get(weather_effect, 'SPECIAL')
                                
                                synthetic_metadata_entry = {
                                    'image_id': synthetic_id,
                                    'filename': f"{synthetic_id}.jpg",
                                    'base_image_id': base_metadata['image_id'],
                                    'source_dataset': 'synthetic',
                                    'weather_condition': weather_label,
                                    'weather_effect': weather_effect,
                                    'is_synthetic': True,
                                    'base_weather': base_metadata['weather_condition'],
                                    'modality': 'rgb',
                                    'standardized_size': self.target_size,
                                    'file_size_mb': file_size_mb,
                                    'quality_score': self.calculate_quality_score(synthetic_img)
                                }
                                
                                synthetic_metadata.append(synthetic_metadata_entry)
                                
                                # Update statistics
                                self.processing_stats['total_processed'] += 1
                                self.processing_stats['synthetic_images'] += 1
                                self.processing_stats['by_weather_type'][weather_label] += 1
                                self.processing_stats['file_sizes_mb'].append(file_size_mb)
                                
                                generated_count += 1
                                pbar.update(1)
                                
                            except Exception as e:
                                print(f"⚠️ Failed to generate {weather_effect} for {base_metadata['image_id']}: {e}")
                                self.processing_stats['failed_processing'] += 1
                
                except Exception as e:
                    print(f"⚠️ Failed to load base image {base_img_path}: {e}")
                    continue
        
        print(f"✅ SYNTHETIC GENERATION COMPLETE: {len(synthetic_metadata):,} images created")
        return synthetic_metadata

    def create_train_val_split(self, all_metadata, train_ratio=0.8):
        """Create balanced train/validation split."""
        # Group by weather condition for balanced split
        weather_groups = defaultdict(list)
        for item in all_metadata:
            weather_groups[item['weather_condition']].append(item['image_id'])
        
        train_ids = []
        val_ids = []
        
        for weather, ids in weather_groups.items():
            random.shuffle(ids)
            split_point = int(len(ids) * train_ratio)
            train_ids.extend(ids[:split_point])
            val_ids.extend(ids[split_point:])
        
        return train_ids, val_ids
    
    def calculate_dataset_size(self):
        """Calculate total dataset size in GB."""
        total_size = 0
        for file_path in self.images_dir.glob('*.jpg'):
            total_size += file_path.stat().st_size
        
        return total_size / (1024**3)  # Convert to GB

# ==== 3. COMPREHENSIVE SAMPLE VISUALIZATION SYSTEM ====
def create_unified_dataset_sample_superplot(dataset_metadata, samples_per_weather=6, save_path=None):
    """Create the ultimate sample visualization showing real vs synthetic across all weather types."""
    print("🎨 Creating unified dataset sample superplot...")
    
    # Group metadata by weather condition and type (real/synthetic)
    weather_samples = defaultdict(lambda: {'real': [], 'synthetic': []})
    
    for item in dataset_metadata:
        weather = item['weather_condition']
        if item['is_synthetic']:
            weather_samples[weather]['synthetic'].append(item)
        else:
            weather_samples[weather]['real'].append(item)
    
    # Filter weather types that have both real and synthetic samples
    valid_weather_types = [
        weather for weather in weather_samples.keys()
        if len(weather_samples[weather]['real']) >= samples_per_weather and 
           len(weather_samples[weather]['synthetic']) >= samples_per_weather
    ]
    
    if not valid_weather_types:
        print("⚠️ Not enough samples for comprehensive visualization")
        return False
    
    # Create mega sample grid
    num_weather = len(valid_weather_types)
    fig, axes = plt.subplots(num_weather, samples_per_weather * 2, figsize=(30, 6 * num_weather))
    
    if num_weather == 1:
        axes = axes.reshape(1, -1)
    
    fig.suptitle('🌦️ UNIFIED DATASET SUPERPLOT: REAL vs SYNTHETIC COMPARISON', 
                fontsize=20, fontweight='bold', y=0.98)
    
    for row, weather_type in enumerate(valid_weather_types):
        # Sample real images
        real_samples = random.sample(weather_samples[weather_type]['real'], samples_per_weather)
        synthetic_samples = random.sample(weather_samples[weather_type]['synthetic'], samples_per_weather)
        
        # Plot real samples (left half)
        for col in range(samples_per_weather):
            try:
                img_path = UNIFIED_DATASET_DIR / 'images' / real_samples[col]['filename']
                with Image.open(img_path) as img:
                    axes[row, col].imshow(np.array(img))
                    axes[row, col].set_title(f'REAL {weather_type}\nID: {real_samples[col]["image_id"]}', 
                                           fontsize=10, color='blue')
                    axes[row, col].axis('off')
            except Exception as e:
                axes[row, col].text(0.5, 0.5, f'Load\nError', ha='center', va='center',
                                   transform=axes[row, col].transAxes, fontsize=10, color='red')
                axes[row, col].axis('off')
        
        # Plot synthetic samples (right half)
        for col in range(samples_per_weather):
            synthetic_col = col + samples_per_weather
            try:
                img_path = UNIFIED_DATASET_DIR / 'images' / synthetic_samples[col]['filename']
                with Image.open(img_path) as img:
                    axes[row, synthetic_col].imshow(np.array(img))
                    
                    effect_name = synthetic_samples[col].get('weather_effect', 'unknown')
                    axes[row, synthetic_col].set_title(f'SYNTHETIC {weather_type}\n{effect_name.replace("_", " ").title()}', 
                                                     fontsize=10, color='green')
                    axes[row, synthetic_col].axis('off')
            except Exception as e:
                axes[row, synthetic_col].text(0.5, 0.5, f'Load\nError', ha='center', va='center',
                                             transform=axes[row, synthetic_col].transAxes, fontsize=10, color='red')
                axes[row, synthetic_col].axis('off')
        
        # Add weather type label on the left
        if samples_per_weather * 2 > 0:
            fig.text(0.02, 0.95 - (row / num_weather), f'{weather_type}', 
                    fontsize=16, fontweight='bold', rotation=90, va='center', ha='center',
                    bbox=dict(boxstyle="round,pad=0.3", facecolor="yellow", alpha=0.8))
    
    # Add column headers
    fig.text(0.3, 0.99, 'REAL IMAGES', fontsize=18, fontweight='bold', ha='center', color='blue')
    fig.text(0.7, 0.99, 'SYNTHETIC IMAGES', fontsize=18, fontweight='bold', ha='center', color='green')
    
    plt.tight_layout()
    
    if save_path:
        plt.savefig(save_path, dpi=200, bbox_inches='tight', facecolor='white')
        print(f"🎨 Sample superplot saved: {save_path}")
    
    plt.show()
    return True

        train_loader = DataLoader(

# ==== 2. ENHANCED MOBILEVIT BACKBONE WITH MULTI-MODAL FUSION ====
class EnhancedMobileViTBackbone(nn.Module):
    """Enhanced MobileViT backbone with proper RGB+NIR fusion and channel alignment."""
    
    def __init__(self, model_name='mobilevit_s', pretrained=True, fusion_method='adaptive'):
        super().__init__()
        self.fusion_method = fusion_method
        
        # Load pre-trained MobileViT
        self.rgb_backbone = timm.create_model(
            model_name, 
            pretrained=pretrained, 
            features_only=True, 
            out_indices=[1, 2, 3, 4]  # Multi-scale features
        )
        
        # Get feature channel dimensions with proper error handling
        with torch.no_grad():
            try:
                dummy_input = torch.randn(1, 3, 512, 512)
                features = self.rgb_backbone(dummy_input)
                self.feature_channels = [f.shape[1] for f in features]
                print(f"✅ RGB Backbone loaded: {model_name}")
                print(f"📊 Feature channels: {self.feature_channels}")
            except Exception as e:
                print(f"⚠️ Error getting feature channels: {e}")
                # Default channels for mobilevit_s
                self.feature_channels = [64, 80, 96, 384]
        
        # NIR processing branch (lightweight)
        self.nir_backbone = nn.Sequential(
            nn.Conv2d(1, 16, 3, padding=1, bias=False),
            nn.BatchNorm2d(16),
            nn.ReLU(inplace=True),
            nn.Conv2d(16, 32, 3, padding=1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(inplace=True),
            nn.AdaptiveAvgPool2d((1, 1)),
            nn.Flatten(),
            nn.Linear(32, 64),
            nn.ReLU(inplace=True)
        )
        
        # ✅ CRITICAL FIX: Proper fusion convolutions that maintain channel dimensions
        self.fusion_convs = nn.ModuleList()
        for i, channels in enumerate(self.feature_channels):
            if fusion_method == 'adaptive':
                # Maintain exact same channel count
                fusion_conv = nn.Sequential(
                    nn.Conv2d(channels, channels, 1, bias=False),
                    nn.BatchNorm2d(channels),
                    nn.ReLU(inplace=True),
                    nn.Conv2d(channels, channels, 3, padding=1, bias=False),
                    nn.BatchNorm2d(channels),
                    nn.ReLU(inplace=True)
                )
            else:
                # Channel attention mechanism
                fusion_conv = nn.Sequential(
                    nn.AdaptiveAvgPool2d(1),
                    nn.Conv2d(channels, max(channels // 16, 1), 1),
                    nn.ReLU(inplace=True),
                    nn.Conv2d(max(channels // 16, 1), channels, 1),
                    nn.Sigmoid()
                )
            
            self.fusion_convs.append(fusion_conv)
        
        # NIR feature projection for adaptive fusion
        if fusion_method == 'adaptive':
            self.nir_projections = nn.ModuleList([
                nn.Linear(64, channels) for channels in self.feature_channels
            ])
        
        print(f"🔗 Multi-modal fusion setup: {fusion_method}")
    
    def forward(self, rgb, nir=None):
        """Forward pass with optional NIR fusion."""
        # RGB feature extraction
        rgb_features = self.rgb_backbone(rgb)
        
        # If no NIR or NIR is None/invalid, return RGB features directly
        if nir is None:
            return rgb_features
        
        # Handle NIR input validation
        try:
            if isinstance(nir, (list, tuple)):
                # Convert list/tuple to tensor
                valid_nir = []
                for n in nir:
                    if n is not None and torch.is_tensor(n):
                        valid_nir.append(n)
                    else:
                        valid_nir.append(torch.zeros_like(rgb[:1, :1]))  # Dummy NIR
                
                if valid_nir:
                    nir = torch.stack(valid_nir, dim=0).squeeze()
                else:
                    return rgb_features
            
            # Ensure NIR has correct dimensions
            if nir.dim() == 3:  # [C, H, W]
                nir = nir.unsqueeze(0)  # Add batch dimension
            
            if nir.shape[1] != 1:  # Convert to single channel if needed
                nir = nir.mean(dim=1, keepdim=True)
            
            # NIR feature extraction
            nir_global = self.nir_backbone(nir)  # Global NIR features [B, 64]
            
        except Exception as e:
            print(f"⚠️ NIR processing failed: {e}, using RGB-only")
            return rgb_features
        
        # Fuse RGB and NIR features
        fused_features = []
        
        for i, (rgb_feat, fusion_conv) in enumerate(zip(rgb_features, self.fusion_convs)):
            if self.fusion_method == 'adaptive':
                try:
                    # Project NIR features to match RGB feature dimensions
                    nir_proj = self.nir_projections[i](nir_global)  # [B, C]
                    nir_influence = nir_proj.unsqueeze(-1).unsqueeze(-1)  # [B, C, 1, 1]
                    
                    # Apply NIR influence to RGB features (additive enhancement)
                    enhanced_rgb = rgb_feat + rgb_feat * nir_influence * 0.1  # Scale influence
                    fused = fusion_conv(enhanced_rgb)  # Maintains channel count
                    
                except Exception as e:
                    print(f"⚠️ Adaptive fusion failed at level {i}: {e}")
                    fused = rgb_feat  # Fallback to RGB only
                    
            else:  # Channel attention fusion
                try:
                    attention = fusion_conv(rgb_feat)
                    fused = rgb_feat * attention
                except Exception as e:
                    print(f"⚠️ Attention fusion failed at level {i}: {e}")
                    fused = rgb_feat  # Fallback to RGB only
            
            fused_features.append(fused)
        
        return fused_features

# ==== 3. WEATHER PROTOTYPICAL CLASSIFICATION HEAD ====
class WeatherPrototypicalClassifier(nn.Module):
    """Advanced weather classifier using prototypical networks for few-shot learning."""
    
    def __init__(self, input_channels=384, embedding_dim=512, num_classes=6):
        super().__init__()
        self.num_classes = num_classes
        self.embedding_dim = embedding_dim
        
        # Feature encoder with proper dimensionality handling
        self.encoder = nn.Sequential(
            nn.AdaptiveAvgPool2d(4),  # Reduce spatial dimensions
            nn.Flatten(),
            nn.Linear(input_channels * 16, embedding_dim * 2),
            nn.ReLU(inplace=True),
            nn.Dropout(0.3),
            nn.Linear(embedding_dim * 2, embedding_dim),
            nn.ReLU(inplace=True),
            nn.Dropout(0.2)
        )
        
        # Learnable prototypes (class representatives)
        self.prototypes = nn.Parameter(torch.randn(num_classes, embedding_dim))
        nn.init.xavier_uniform_(self.prototypes)
        
        # Temperature parameter for softmax scaling
        self.temperature = nn.Parameter(torch.tensor(10.0))
        
        print(f"🌦️ Weather Prototypical Classifier: {num_classes} classes, {embedding_dim}D embeddings")
    
    def forward(self, features, return_embeddings=False):
        """Forward pass with prototype-based classification."""
        # Extract embeddings
        embeddings = self.encoder(features)  # [B, embedding_dim]
        
        if return_embeddings:
            return embeddings
        
        # Compute distances to prototypes (Euclidean distance)
        # embeddings: [B, embedding_dim], prototypes: [num_classes, embedding_dim]
        distances = torch.cdist(embeddings.unsqueeze(1), self.prototypes.unsqueeze(0))  # [B, 1, num_classes]
        distances = distances.squeeze(1)  # [B, num_classes]
        
        # Convert distances to similarities (negative distance)
        similarities = -distances / torch.clamp(self.temperature, min=0.1)
        
        return similarities
    
    def update_prototypes(self, embeddings, labels):
        """Update prototypes using support samples (for meta-learning)."""
        with torch.no_grad():
            for class_id in range(self.num_classes):
                class_mask = (labels == class_id)
                if class_mask.sum() > 0:
                    class_embeddings = embeddings[class_mask]
                    prototype_update = class_embeddings.mean(dim=0)
                    # Exponential moving average update
                    momentum = 0.9
                    self.prototypes[class_id] = momentum * self.prototypes[class_id] + (1 - momentum) * prototype_update
    
    def get_prototype_distances(self, embeddings):
        """Get distances to all prototypes for analysis."""
        distances = torch.cdist(embeddings.unsqueeze(1), self.prototypes.unsqueeze(0))
        return distances.squeeze(1)

# ==== 4. SEGMENTATION HEAD ====
class SegmentationHead(nn.Module):
    """Advanced segmentation head with FPN-style multi-scale feature fusion."""
    
    def __init__(self, feature_channels, num_classes=21, output_stride=4):
        super().__init__()
        self.output_stride = output_stride
        self.num_classes = num_classes
        
        # FPN-style decoder
        self.lateral_convs = nn.ModuleList()
        self.fpn_convs = nn.ModuleList()
        
        fpn_dim = 256
        
        for channels in reversed(feature_channels):  # Start from highest level
            # Lateral connection to reduce channels
            lateral_conv = nn.Conv2d(channels, fpn_dim, 1, bias=False)
            
            # Feature processing after fusion
            fpn_conv = nn.Sequential(
                nn.Conv2d(fpn_dim, fpn_dim, 3, padding=1, bias=False),
                nn.BatchNorm2d(fpn_dim),
                nn.ReLU(inplace=True)
            )
            
            self.lateral_convs.append(lateral_conv)
            self.fpn_convs.append(fpn_conv)
        
        # Final classifier
        self.classifier = nn.Sequential(
            nn.Conv2d(fpn_dim, fpn_dim // 2, 3, padding=1, bias=False),
            nn.BatchNorm2d(fpn_dim // 2),
            nn.ReLU(inplace=True),
            nn.Dropout2d(0.1),
            nn.Conv2d(fpn_dim // 2, num_classes, 1)
        )
        
        print(f"🎯 Segmentation Head: FPN with {num_classes} classes, {fpn_dim} FPN channels")
    
    def forward(self, features):
        """Forward pass with multi-scale feature fusion."""
        # Build FPN from top to bottom
        fpn_features = []
        prev_feature = None
        
        for i, (feature, lateral_conv, fpn_conv) in enumerate(
            zip(reversed(features), self.lateral_convs, self.fpn_convs)
        ):
            # Lateral connection
            lateral = lateral_conv(feature)
            
            if prev_feature is not None:
                # Upsample previous feature and add
                upsampled = F.interpolate(
                    prev_feature, size=lateral.shape[-2:], 
                    mode='bilinear', align_corners=False
                )
                lateral = lateral + upsampled
            
            # Process fused feature
            fpn_feature = fpn_conv(lateral)
            fpn_features.append(fpn_feature)
            prev_feature = fpn_feature
        
        # Use the finest scale feature for segmentation
        seg_feature = fpn_features[-1]  # Finest scale (highest resolution)
        
        # Generate segmentation logits
        seg_logits = self.classifier(seg_feature)
        
        # Upsample to input resolution
        seg_logits = F.interpolate(
            seg_logits, scale_factor=self.output_stride, 
            mode='bilinear', align_corners=False
        )
        
        return seg_logits

# ==== 5. DETECTION HEAD ====
class DetectionHead(nn.Module):
    """Multi-scale detection head for traffic objects."""
    
    def __init__(self, feature_channels, num_classes=1, num_anchors=3):
        super().__init__()
        self.num_classes = num_classes
        self.num_anchors = num_anchors
        
        # Use multiple scales for detection (last 3 feature levels)
        self.detection_heads = nn.ModuleList()
        
        for channels in feature_channels[-3:]:  # Use last 3 scales
            head = nn.Sequential(
                nn.Conv2d(channels, channels, 3, padding=1, bias=False),
                nn.BatchNorm2d(channels),
                nn.ReLU(inplace=True),
                nn.Conv2d(channels, channels, 3, padding=1, bias=False),
                nn.BatchNorm2d(channels),
                nn.ReLU(inplace=True),
                # Output: [x, y, w, h, objectness] + class_scores
                nn.Conv2d(channels, num_anchors * (5 + num_classes), 1)
            )
            self.detection_heads.append(head)
        
        print(f"🔍 Detection Head: {len(self.detection_heads)} scales, {num_classes} classes, {num_anchors} anchors")
    
    def forward(self, features):
        """Forward pass for multi-scale detection."""
        detections = []
        
        for i, (feature, head) in enumerate(zip(features[-3:], self.detection_heads)):
            try:
                detection = head(feature)
                
                # Reshape: [B, anchors*(5+classes), H, W] -> [B, anchors, H, W, 5+classes]
                B, _, H, W = detection.shape
                detection = detection.view(B, self.num_anchors, 5 + self.num_classes, H, W)
                detection = detection.permute(0, 1, 3, 4, 2).contiguous()
                
                detections.append(detection)
                
            except Exception as e:
                print(f"⚠️ Detection head {i} failed: {e}")
                # Return dummy detection
                B = feature.shape[0]
                H, W = feature.shape[-2:]
                dummy_detection = torch.zeros(B, self.num_anchors, H, W, 5 + self.num_classes).to(feature.device)
                detections.append(dummy_detection)
        
        return detections

            input_channels=feature_channels[-1],  # Use highest level features
            embedding_dim=512,
            num_classes=num_weather_classes
        )
        
        self.segmentation_head = SegmentationHead(
            feature_channels=feature_channels,
            num_classes=num_seg_classes
        )
        
        self.detection_head = DetectionHead(
            feature_channels=feature_channels,
            num_classes=num_det_classes
        )
        
        # Weather-aware feature gating mechanism
        self.weather_gates = nn.ModuleList([
            nn.Sequential(
                nn.AdaptiveAvgPool2d(1),
                nn.Flatten(),
                nn.Linear(channels, num_weather_classes),
                nn.Softmax(dim=1)
            ) for channels in feature_channels
        ])
        
        # Initialize weights
        self._initialize_weights()
        
        print("✅ Unified Multi-Task ADAS Model Initialized Successfully!")
        print(f"🌦️ Weather Classes: {num_weather_classes}")
        print(f"🎯 Segmentation Classes: {num_seg_classes}")
        print(f"🔍 Detection Classes: {num_det_classes}")
        print(f"🧠 MAML Support: {'Enabled' if enable_maml else 'Disabled'}")
    
    def _initialize_weights(self):
        """Initialize model weights properly."""
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
                if m.bias is not None:
                    nn.init.constant_(m.bias, 0)
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.constant_(m.weight, 1)
                nn.init.constant_(m.bias, 0)
            elif isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, 0, 0.01)
                if m.bias is not None:
                    nn.init.constant_(m.bias, 0)
    
    def forward(self, rgb, nir=None, return_features=False):
        """Forward pass through all tasks with comprehensive error handling."""
        try:
            # Extract multi-scale features
            features = self.backbone(rgb, nir)
            
            if return_features:
                return features
            
            # Weather classification
            weather_logits = self.weather_classifier(features[-1])
            weather_probs = F.softmax(weather_logits, dim=1)
            
            # Segmentation
            seg_logits = self.segmentation_head(features)
            
            # Detection
            det_outputs = self.detection_head(features)
            
            # Weather-aware feature gating
            gated_features = []
            for feature, gate in zip(features, self.weather_gates):
                try:
                    gate_weights = gate(feature)  # [B, num_weather_classes]
                    # Apply weather-specific attention (simplified)
                    gate_attention = gate_weights.mean(dim=1, keepdim=True).unsqueeze(-1).unsqueeze(-1)
                    gated_feature = feature * (0.8 + 0.2 * gate_attention)  # Gentle gating
                    gated_features.append(gated_feature)
                except Exception as e:
                    print(f"⚠️ Weather gating failed: {e}")
                    gated_features.append(feature)  # Fallback to original feature
            
            return {
                'weather_logits': weather_logits,
                'weather_probs': weather_probs,
                'segmentation': seg_logits,
                'detection': det_outputs,
                'features': features,
                'gated_features': gated_features
            }
            
        except Exception as e:
            print(f"⚠️ Model forward pass failed: {e}")
            # Return dummy outputs to prevent training crashes
            batch_size = rgb.shape[0]
            device = rgb.device
            
            return {
                'weather_logits': torch.zeros(batch_size, 6).to(device),
                'weather_probs': torch.ones(batch_size, 6).to(device) / 6,
                'segmentation': torch.zeros(batch_size, 21, 512, 512).to(device),
                'detection': [torch.zeros(batch_size, 3, 16, 16, 6).to(device)],
                'features': [torch.zeros(batch_size, 64, 128, 128).to(device)],
                'gated_features': [torch.zeros(batch_size, 64, 128, 128).to(device)]
            }
    
    def get_weather_embeddings(self, rgb, nir=None):
        """Get weather embeddings for prototypical learning and analysis."""
        try:
            features = self.backbone(rgb, nir)
            embeddings = self.weather_classifier(features[-1], return_embeddings=True)
            return embeddings
        except Exception as e:
            print(f"⚠️ Weather embedding extraction failed: {e}")
            return torch.zeros(rgb.shape[0], 512).to(rgb.device)
    
    def adapt_to_weather(self, support_rgb, support_nir, support_labels):
        """Adapt weather classifier using prototypical learning."""
        try:
            # Get embeddings for support samples
            support_embeddings = self.get_weather_embeddings(support_rgb, support_nir)
            
            # Update prototypes
            self.weather_classifier.update_prototypes(support_embeddings, support_labels)
            
            return True
        except Exception as e:
            print(f"⚠️ Weather adaptation failed: {e}")
            return False

        self.loss_weights = loss_weights
        self.device = device
        
        # ✅ AUTOCAST-SAFE LOSS FUNCTIONS
        self.weather_loss = nn.CrossEntropyLoss(label_smoothing=0.1)
        self.seg_loss = nn.CrossEntropyLoss(ignore_index=255)
        
        # ✅ CRITICAL FIX: Use BCEWithLogitsLoss instead of BCE for autocast safety
        self.detection_loss_fn = nn.BCEWithLogitsLoss()
        
        # Learnable adaptive weights
        self.adaptive_weights = nn.Parameter(
            torch.tensor(list(loss_weights.values()), dtype=torch.float32)
        ).to(device)
        
        # Loss history tracking
        self.loss_history = {task: [] for task in loss_weights.keys()}
        self.loss_trends = {task: 0.0 for task in loss_weights.keys()}
        
        print(f"📊 Loss manager initialized with adaptive weighting")
    
    def compute_detection_loss_safe(self, predictions, stage='full_multitask'):
        """Autocast-safe detection loss computation."""
        if not predictions or stage not in ['full_multitask', 'fine_tuning']:
            return torch.tensor(0.0, device=self.device)
        
        total_loss = torch.tensor(0.0, device=self.device)
        valid_predictions = 0
        
        for pred_scale in predictions:
            if pred_scale.numel() == 0:
                continue
            
            try:
                # Extract confidence logits (NOT sigmoid outputs)
                pred_conf_logits = pred_scale[..., 4]  # Raw logits
                
                # Create dummy targets (you can enhance this with real detection targets)
                target_conf = torch.ones_like(pred_conf_logits) * 0.5
                
                # ✅ AUTOCAST-SAFE: Use BCEWithLogitsLoss (combines sigmoid + BCE)
                conf_loss = self.detection_loss_fn(pred_conf_logits, target_conf)
                total_loss += conf_loss
                valid_predictions += 1
                
            except Exception as e:
                print(f"⚠️ Detection loss computation failed for scale: {e}")
                continue
        
        # Average over valid predictions
        if valid_predictions > 0:
            total_loss = total_loss / valid_predictions
        
        return total_loss
    
    def compute_losses(self, outputs, targets, stage='multitask_pretrain'):
        """Compute all task losses with comprehensive error handling."""
        losses = {}
        total_loss = torch.tensor(0.0, device=self.device)
        active_tasks = config.stages[stage].get('tasks', ['weather'])
        
        # Weather classification loss (always computed)
        if 'weather_logits' in outputs and 'weather' in active_tasks:
            try:
                weather_targets = targets['weather_label']
                weather_loss = self.weather_loss(outputs['weather_logits'], weather_targets)
                losses['weather'] = weather_loss
                total_loss += self.adaptive_weights[0] * weather_loss
            except Exception as e:
                print(f"⚠️ Weather loss computation failed: {e}")
                losses['weather'] = torch.tensor(0.0, device=self.device)
        
        # Segmentation loss
        if 'segmentation' in outputs and 'segmentation' in active_tasks:
            try:
                # Create dummy segmentation targets (enhance this with real targets if available)
                batch_size = outputs['segmentation'].shape[0]
                seg_targets = torch.zeros(batch_size, 512, 512, dtype=torch.long, device=self.device)
                
                seg_loss = self.seg_loss(outputs['segmentation'], seg_targets)
                losses['segmentation'] = seg_loss
                total_loss += self.adaptive_weights[1] * seg_loss
                
            except Exception as e:
                print(f"⚠️ Segmentation loss computation failed: {e}")
                losses['segmentation'] = torch.tensor(0.0, device=self.device)
        
        # Detection loss
        if 'detection' in outputs and 'detection' in active_tasks:
            try:
                det_loss = self.compute_detection_loss_safe(outputs['detection'], stage)
                losses['detection'] = det_loss
                total_loss += self.adaptive_weights[2] * det_loss
                
            except Exception as e:
                print(f"⚠️ Detection loss computation failed: {e}")
                losses['detection'] = torch.tensor(0.0, device=self.device)
        
        losses['total'] = total_loss
        return losses
    
    def update_adaptive_weights(self):
        """Update adaptive loss weights based on loss trends."""
        if all(len(self.loss_history[task]) < 3 for task in self.loss_history.keys()):
            return
        
        with torch.no_grad():
            for i, task in enumerate(['weather', 'segmentation', 'detection']):
                if task in self.loss_history and len(self.loss_history[task]) >= 3:
                    recent_losses = self.loss_history[task][-3:]
                    
                    # Compute trend (positive = increasing loss, needs more attention)
                    if len(recent_losses) >= 2:
                        trend = (recent_losses[-1] - recent_losses[0]) / (recent_losses[0] + 1e-8)
                        self.loss_trends[task] = trend
                        
                        # Adjust weight (increase if loss is increasing)
                        adjustment = 1.0 + 0.05 * trend  # Gentle adjustment
                        self.adaptive_weights.data[i] *= adjustment
                        self.adaptive_weights.data[i] = torch.clamp(self.adaptive_weights.data[i], 0.1, 2.0)

# ==== 3. ENHANCED MAML META-LEARNING TRAINER ====
class EnhancedMAMLTrainer:
    """Production-ready MAML trainer with comprehensive error handling."""
    
    def __init__(self, model, inner_lr=0.01, meta_lr=1e-3, num_inner_steps=3):
        self.model = model
        self.inner_lr = inner_lr
        self.meta_lr = meta_lr
        self.num_inner_steps = num_inner_steps
        
        # Meta optimizer (only weather classifier for efficiency)
        self.meta_optimizer = optim.Adam(
            self.model.weather_classifier.parameters(),
            lr=meta_lr,
            weight_decay=1e-5
        )
        
        self.meta_loss_history = []
        self.adaptation_success_rate = []
        
        print(f"🧠 MAML trainer initialized (inner_lr={inner_lr}, meta_lr={meta_lr})")
    
    def create_meta_task_batch(self, dataloader, meta_batch_size=3, support_shots=3, query_shots=2):
        """Create meta-learning tasks with proper error handling."""
        meta_tasks = []
        
        # Collect samples by weather type
        weather_samples = defaultdict(list)
        samples_collected = 0
        max_samples_per_weather = 20  # Limit to prevent memory issues
        
        print(f"🔍 Collecting samples for MAML meta-tasks...")
        
        for batch_idx, batch in enumerate(dataloader):
            if samples_collected >= meta_batch_size * 50:  # Enough samples
                break
            
            try:
                for i in range(len(batch['weather_name'])):
                    weather = batch['weather_name'][i]
                    if len(weather_samples[weather]) < max_samples_per_weather:
                        sample = {
                            'rgb': batch['rgb'][i].to(device),
                            'nir': batch['nir'][i].to(device) if batch['nir'][i] is not None else torch.zeros(1, 512, 512).to(device),
                            'weather_label': batch['weather_label'][i].to(device),
                            'weather_name': weather
                        }
                        weather_samples[weather].append(sample)
                        samples_collected += 1
                
            except Exception as e:
                print(f"⚠️ Error collecting batch {batch_idx}: {e}")
                continue
            
            if batch_idx >= 20:  # Limit data collection
                break
        
        print(f"📊 Collected samples: {dict([(w, len(s)) for w, s in weather_samples.items()])}")
        
        # Create meta-tasks
        for task_id in range(meta_batch_size):
            try:
                # Select weather types with enough samples
                valid_weathers = [
                    w for w, samples in weather_samples.items() 
                    if len(samples) >= support_shots + query_shots
                ]
                
                if len(valid_weathers) < 2:
                    print(f"⚠️ Not enough weather types for meta-task {task_id}")
                    continue
                
                # Sample weather types for this task
                task_weathers = random.sample(valid_weathers, min(3, len(valid_weathers)))
                
                support_samples = []
                query_samples = []
                
                for weather in task_weathers:
                    available_samples = weather_samples[weather]
                    selected_indices = random.sample(
                        range(len(available_samples)), 
                        min(support_shots + query_shots, len(available_samples))
                    )
                    
                    # Split into support and query
                    support_indices = selected_indices[:support_shots]
                    query_indices = selected_indices[support_shots:support_shots + query_shots]
                    
                    for idx in support_indices:
                        support_samples.append(available_samples[idx])
                    
                    for idx in query_indices:
                        query_samples.append(available_samples[idx])
                
                if len(support_samples) >= 3 and len(query_samples) >= 2:
                    # Convert to proper format
                    support_set = {
                        'rgb': torch.stack([s['rgb'] for s in support_samples]),
                        'nir': torch.stack([s['nir'] for s in support_samples]),
                        'labels': torch.stack([s['weather_label'] for s in support_samples])
                    }
                    
                    query_set = {
                        'rgb': torch.stack([s['rgb'] for s in query_samples]),
                        'nir': torch.stack([s['nir'] for s in query_samples]),
                        'labels': torch.stack([s['weather_label'] for s in query_samples])
                    }
                    
                    meta_tasks.append({
                        'support': support_set,
                        'query': query_set,
                        'task_id': task_id
                    })
                
            except Exception as e:
                print(f"⚠️ Error creating meta-task {task_id}: {e}")
                continue
        
        print(f"✅ Created {len(meta_tasks)} meta-tasks for MAML training")
        return meta_tasks
    
    def meta_training_step(self, meta_tasks):
        """Perform MAML meta-training step with Higher if available."""
        if not FEATURE_FLAGS.get('higher', False):
            print("⚠️ Higher not available, skipping MAML training")
            return {'meta_loss': 0.0, 'adaptation_success': False}
        
        if not meta_tasks:
            return {'meta_loss': 0.0, 'adaptation_success': False}
        
        try:
            import higher
            
            meta_losses = []
            successful_adaptations = 0
            
            for task in meta_tasks:
                try:
                    support_set = task['support']
                    query_set = task['query']
                    
                    # MAML inner loop with Higher
                    with higher.innerloop_ctx(
                        self.model.weather_classifier,
                        optim.SGD(self.model.weather_classifier.parameters(), lr=self.inner_lr),
                        copy_initial_weights=False
                    ) as (fmodel, diffopt):
                        
                        # Inner loop adaptation
                        for step in range(self.num_inner_steps):
                            # Get features from backbone (frozen during inner loop)
                            with torch.no_grad():
                                support_features = self.model.backbone(support_set['rgb'], support_set['nir'])
                            
                            # Adapted classifier forward pass
                            support_logits = fmodel(support_features[-1])
                            support_loss = F.cross_entropy(support_logits, support_set['labels'])
                            
                            # Inner gradient update
                            diffopt.step(support_loss)
                        
                        # Query evaluation (meta-gradient computation)
                        with torch.no_grad():
                            query_features = self.model.backbone(query_set['rgb'], query_set['nir'])
                        
                        query_logits = fmodel(query_features[-1])
                        query_loss = F.cross_entropy(query_logits, query_set['labels'])
                        
                        meta_losses.append(query_loss)
                        successful_adaptations += 1
                
                except Exception as e:
                    print(f"⚠️ MAML task failed: {e}")
                    continue
            
            # Meta-optimization step
            if meta_losses:
                meta_loss = torch.stack(meta_losses).mean()
                
                self.meta_optimizer.zero_grad()
                meta_loss.backward()
                torch.nn.utils.clip_grad_norm_(self.model.weather_classifier.parameters(), 1.0)
                self.meta_optimizer.step()
                
                self.meta_loss_history.append(meta_loss.item())
                success_rate = successful_adaptations / len(meta_tasks)
                self.adaptation_success_rate.append(success_rate)
                
                return {
                    'meta_loss': meta_loss.item(),
                    'adaptation_success': True,
                    'success_rate': success_rate,
                    'num_tasks': len(meta_tasks)
                }
            
        except Exception as e:
            print(f"⚠️ MAML meta-training failed: {e}")
        
        return {'meta_loss': 0.0, 'adaptation_success': False}

# ==== 4. COMPREHENSIVE TRAINING MANAGER ====
class ProgressiveTrainingManager:
    """Ultimate training manager with comprehensive monitoring and error handling."""
    
    def __init__(self, model, train_loader, val_loader, config, save_dir):
        self.model = model
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.config = config
        self.save_dir = Path(save_dir)
        
        # Initialize components
        self.loss_manager = AutocastSafeMultiTaskLossManager(config.initial_loss_weights, device)
        self.maml_trainer = EnhancedMAMLTrainer(model) if FEATURE_FLAGS.get('higher', False) else None
        self.scaler = GradScaler() if config.mixed_precision else None
        
        # Training state tracking
        self.training_history = {
            'stages': {},
            'best_metrics': {},
            'checkpoints': [],
            'training_curves': {
                'train_loss': [],
                'val_loss': [],
                'weather_acc': [],
                'learning_rates': []
            }
        }
        
        self.current_stage = None
        self.global_step = 0
        self.best_overall_metric = 0.0
        
        print(f"🎯 Training manager initialized")
        print(f"💾 Checkpoints will be saved to: {save_dir}")
    
    def train_progressive_stages(self):
        """Execute all training stages progressively."""
        print("\n🚀 STARTING PROGRESSIVE MULTI-STAGE TRAINING")
        print("="*80)
        
        stage_results = {}
        
        for stage_name, stage_config in self.config.stages.items():
            print(f"\n🔥 STAGE: {stage_name.upper()}")
            print(f"📋 {stage_config['description']}")
            
            try:
                best_metric = self.train_single_stage(stage_name, stage_config)
                stage_results[stage_name] = best_metric
                
                print(f"✅ Stage {stage_name} completed with best metric: {best_metric:.4f}")
                
                # Save stage checkpoint
                self.save_stage_checkpoint(stage_name, best_metric)
                
            except Exception as e:
                print(f"❌ Stage {stage_name} failed: {e}")
                stage_results[stage_name] = 0.0
                continue
        
        print("\n🎉 PROGRESSIVE TRAINING COMPLETED!")
        print("="*80)
        print("📊 FINAL RESULTS:")
        for stage, metric in stage_results.items():
            print(f"   • {stage.replace('_', ' ').title()}: {metric:.4f}")
        
        return stage_results
    
    def train_single_stage(self, stage_name, stage_config):
        """Train a single stage with comprehensive monitoring."""
        self.current_stage = stage_name
        
        # Stage-specific setup
        optimizer = self.create_stage_optimizer(stage_config)
        scheduler = self.create_stage_scheduler(optimizer, stage_config)
        
        # Apply stage-specific model configuration
        self.configure_model_for_stage(stage_config)
        
        best_stage_metric = 0.0
        epochs_without_improvement = 0
        
        print(f"⚙️ Stage config: LR={stage_config['lr']}, Tasks={stage_config.get('tasks', ['weather'])}")
        
        for epoch in range(stage_config['epochs']):
            print(f"\n📅 Epoch {epoch+1}/{stage_config['epochs']} - Stage: {stage_name}")
            
            # Training epoch
            train_metrics = self.train_epoch(stage_config, optimizer, epoch)
            
            # Validation epoch
            val_metrics = self.validate_epoch(stage_config, epoch)
            
            # Learning rate scheduling
            if scheduler:
                if isinstance(scheduler, ReduceLROnPlateau):
                    scheduler.step(val_metrics.get('weather_acc', 0))
                else:
                    scheduler.step()
            
            # Track training curves
            self.training_history['training_curves']['train_loss'].append(train_metrics.get('total', 0))
            self.training_history['training_curves']['val_loss'].append(val_metrics.get('total', 0))
            self.training_history['training_curves']['weather_acc'].append(val_metrics.get('weather_acc', 0))
            self.training_history['training_curves']['learning_rates'].append(optimizer.param_groups[0]['lr'])
            
            # Compute composite metric
            current_metric = val_metrics.get('weather_acc', 0) * 0.6 + (1.0 - val_metrics.get('total', 1.0)) * 0.4
            
            # Best model tracking
            if current_metric > best_stage_metric:
                best_stage_metric = current_metric
                epochs_without_improvement = 0
                self.save_best_checkpoint(stage_name, epoch, optimizer, scheduler, val_metrics)
            else:
                epochs_without_improvement += 1
            
            # Early stopping
            if epochs_without_improvement >= self.config.early_stopping_patience:
                print(f"🛑 Early stopping triggered after {epochs_without_improvement} epochs without improvement")
                break
            
            # Memory cleanup
            if epoch % 2 == 0:
                clear_gpu_memory()
        
        return best_stage_metric
    
    def train_epoch(self, stage_config, optimizer, epoch):
        """Train one epoch with comprehensive error handling."""
        self.model.train()
        
        # Apply stage-specific model settings
        if stage_config.get('freeze_backbone', False):
            for param in self.model.backbone.parameters():
                param.requires_grad = False
        else:
            for param in self.model.backbone.parameters():
                param.requires_grad = True
        
        epoch_losses = defaultdict(list)
        num_batches = 0
        
        # Progress bar with rich information
        progress_bar = tqdm(
            self.train_loader, 
            desc=f"Training {self.current_stage}",
            leave=False
        )
        
        for batch_idx, batch in enumerate(progress_bar):
            try:
                # Move data to device with proper error handling
                rgb = batch['rgb'].to(device, non_blocking=True)
                
                # Handle NIR data safely
                nir = None
                if batch.get('nir') is not None:
                    try:
                        nir_batch = batch['nir']
                        if isinstance(nir_batch, (list, tuple)):
                            nir = torch.stack([
                                n.to(device) if n is not None else torch.zeros(1, 512, 512).to(device)
                                for n in nir_batch
                            ])
                        else:
                            nir = nir_batch.to(device, non_blocking=True)
                    except Exception as e:
                        print(f"⚠️ NIR processing failed: {e}")
                        nir = None
                
                targets = {
                    'weather_label': batch['weather_label'].to(device, non_blocking=True)
                }
                
                # Stage-specific training step
                if self.current_stage == 'maml_adaptation' and self.maml_trainer:
                    batch_losses = self.maml_training_step(batch)
                else:
                    batch_losses = self.standard_training_step(rgb, nir, targets, optimizer, stage_config)
                
                # Accumulate losses
                for loss_name, loss_value in batch_losses.items():
                    if isinstance(loss_value, (int, float)):
                        epoch_losses[loss_name].append(loss_value)
                
                num_batches += 1
                self.global_step += 1
                
                # Update progress bar
                progress_bar.set_postfix({
                    'loss': f"{batch_losses.get('total', 0):.4f}",
                    'weather': f"{batch_losses.get('weather', 0):.4f}",
                    'lr': f"{optimizer.param_groups[0]['lr']:.6f}"
                })
                
                # Periodic logging
                if batch_idx % self.config.log_every == 0:
                    print(f"   Batch {batch_idx}: Loss={batch_losses.get('total', 0):.4f}")
                
                # Memory management
                if batch_idx % 50 == 0:
                    torch.cuda.empty_cache()
                
                # Limit training batches for demonstration (remove for full training)
                if batch_idx >= 150:
                    break
                
            except Exception as e:
                print(f"⚠️ Training batch {batch_idx} failed: {e}")
                continue
        
        # Compute average losses
        avg_losses = {name: np.mean(losses) for name, losses in epoch_losses.items()}
        
        # Update adaptive loss weights
        self.loss_manager.update_adaptive_weights()
        
        print(f"📊 Train Epoch Summary - Total Loss: {avg_losses.get('total', 0):.4f}, Weather Loss: {avg_losses.get('weather', 0):.4f}")
        
        return avg_losses
    
    def standard_training_step(self, rgb, nir, targets, optimizer, stage_config):
        """Standard training step with autocast safety."""
        
        if self.config.mixed_precision and self.scaler:
            # Mixed precision training
            with autocast():
                outputs = self.model(rgb, nir)
                losses = self.loss_manager.compute_losses(outputs, targets, self.current_stage)
                total_loss = losses['total'] / self.config.gradient_accumulation_steps
            
            # Gradient accumulation
            self.scaler.scale(total_loss).backward()
            
            if (self.global_step + 1) % self.config.gradient_accumulation_steps == 0:
                self.scaler.unscale_(optimizer)
                torch.nn.utils.clip_grad_norm_(self.model.parameters(), self.config.max_grad_norm)
                self.scaler.step(optimizer)
                self.scaler.update()
                optimizer.zero_grad()
        
        else:
            # Standard precision training
            outputs = self.model(rgb, nir)
            losses = self.loss_manager.compute_losses(outputs, targets, self.current_stage)
            total_loss = losses['total'] / self.config.gradient_accumulation_steps
            
            total_loss.backward()
            
            if (self.global_step + 1) % self.config.gradient_accumulation_steps == 0:
                torch.nn.utils.clip_grad_norm_(self.model.parameters(), self.config.max_grad_norm)
                optimizer.step()
                optimizer.zero_grad()
        
        return {name: loss.item() if isinstance(loss, torch.Tensor) else loss 
                for name, loss in losses.items()}
    
    def maml_training_step(self, batch):
        """MAML training step for meta-learning."""
        if not self.maml_trainer:
            return {'meta_error': 1.0}
        
        # Create meta-tasks from current batch
        try:
            meta_tasks = self.maml_trainer.create_meta_task_batch([batch], meta_batch_size=2)
            maml_results = self.maml_trainer.meta_training_step(meta_tasks)
            return maml_results
        except Exception as e:
            print(f"⚠️ MAML step failed: {e}")
            return {'meta_error': 1.0}
    
    def validate_epoch(self, stage_config, epoch):
        """Comprehensive validation with multiple metrics."""
        self.model.eval()
        
        val_losses = defaultdict(list)
        weather_correct = 0
        weather_total = 0
        weather_per_class_correct = defaultdict(int)
        weather_per_class_total = defaultdict(int)
        
        with torch.no_grad():
            for batch_idx, batch in enumerate(tqdm(self.val_loader, desc="Validation", leave=False)):
                try:
                    rgb = batch['rgb'].to(device, non_blocking=True)
                    
                    # Handle NIR safely
                    nir = None
                    if batch.get('nir') is not None:
                        try:
                            nir_batch = batch['nir']
                            if isinstance(nir_batch, (list, tuple)):
                                nir = torch.stack([
                                    n.to(device) if n is not None else torch.zeros(1, 512, 512).to(device)
                                    for n in nir_batch
                                ])
                            else:
                                nir = nir_batch.to(device, non_blocking=True)
                        except:
                            nir = None
                    
                    targets = {
                        'weather_label': batch['weather_label'].to(device, non_blocking=True)
                    }
                    
                    # Model forward pass
                    outputs = self.model(rgb, nir)
                    losses = self.loss_manager.compute_losses(outputs, targets, self.current_stage)
                    
                    # Accumulate losses
                    for loss_name, loss_value in losses.items():
                        if isinstance(loss_value, torch.Tensor):
                            val_losses[loss_name].append(loss_value.item())
                        else:
                            val_losses[loss_name].append(loss_value)
                    
                    # Weather classification metrics
                    weather_preds = torch.argmax(outputs['weather_logits'], dim=1)
                    weather_labels = targets['weather_label']
                    
                    # Overall accuracy
                    correct_mask = (weather_preds == weather_labels)
                    weather_correct += correct_mask.sum().item()
                    weather_total += len(weather_labels)
                    
                    # Per-class accuracy
                    for i, (pred, label) in enumerate(zip(weather_preds, weather_labels)):
                        label_name = WEATHER_NAMES[label.item()]
                        weather_per_class_total[label_name] += 1
                        if correct_mask[i]:
                            weather_per_class_correct[label_name] += 1
                    
                    # Limit validation batches for efficiency
                    if batch_idx >= 30:
                        break
                        
                except Exception as e:
                    print(f"⚠️ Validation batch {batch_idx} failed: {e}")
                    continue
        
        # Compute validation metrics
        avg_val_losses = {name: np.mean(losses) for name, losses in val_losses.items()}
        
        weather_acc = weather_correct / max(weather_total, 1)
        per_class_acc = {
            cls: weather_per_class_correct[cls] / max(weather_per_class_total[cls], 1)
            for cls in weather_per_class_total.keys()
        }
        
        val_metrics = {
            'weather_acc': weather_acc,
            'per_class_acc': per_class_acc,
            **avg_val_losses
        }
        
        print(f"📊 Validation Summary:")
        print(f"   Weather Accuracy: {weather_acc:.4f}")
        print(f"   Total Loss: {avg_val_losses.get('total', 0):.4f}")
        print(f"   Per-class Acc: {', '.join([f'{k}:{v:.3f}' for k, v in per_class_acc.items()])}")
        
        return val_metrics
    
    def configure_model_for_stage(self, stage_config):
        """Configure model settings for specific training stage."""
        # Freeze/unfreeze backbone based on stage config
        freeze_backbone = stage_config.get('freeze_backbone', False)
        
        for param in self.model.backbone.parameters():
            param.requires_grad = not freeze_backbone
        
        # Always train weather classifier
        for param in self.model.weather_classifier.parameters():
            param.requires_grad = True
        
        # Configure other heads based on active tasks
        active_tasks = stage_config.get('tasks', ['weather'])
        
        for param in self.model.segmentation_head.parameters():
            param.requires_grad = 'segmentation' in active_tasks
            
        for param in self.model.detection_head.parameters():
            param.requires_grad = 'detection' in active_tasks
        
        print(f"🔧 Model configured - Backbone frozen: {freeze_backbone}, Active tasks: {active_tasks}")
    
    def create_stage_optimizer(self, stage_config):
        """Create optimizer optimized for specific stage."""
        lr = stage_config['lr']
        
        # Collect trainable parameters
        trainable_params = [p for p in self.model.parameters() if p.requires_grad]
        
        if self.current_stage == 'maml_adaptation':
            # Use Adam for MAML with different LR for different components
            param_groups = [
                {'params': self.model.weather_classifier.parameters(), 'lr': lr},
                {'params': [p for n, p in self.model.named_parameters() 
                          if 'weather_classifier' not in n and p.requires_grad], 'lr': lr * 0.1}
            ]
            return optim.Adam(param_groups, weight_decay=1e-5)
        else:
            # Standard AdamW optimizer
            return optim.AdamW(trainable_params, lr=lr, weight_decay=1e-4)
    
    def create_stage_scheduler(self, optimizer, stage_config):
        """Create learning rate scheduler for stage."""
        if self.current_stage == 'warmup':
            return optim.lr_scheduler.LinearLR(optimizer, start_factor=0.1, total_iters=5)
        elif self.current_stage == 'maml_adaptation':
            return optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=stage_config['epochs'])
        elif self.current_stage in ['multitask_pretrain', 'full_multitask']:
            return optim.lr_scheduler.OneCycleLR(
                optimizer, 
                max_lr=stage_config['lr'], 
                epochs=stage_config['epochs'],
                steps_per_epoch=len(self.train_loader)
            )
        else:
            return optim.lr_scheduler.ReduceLROnPlateau(
                optimizer, mode='max', patience=3, factor=0.7, verbose=True
            )
    
    def save_best_checkpoint(self, stage_name, epoch, optimizer, scheduler, metrics):
        """Save best model checkpoint."""
        checkpoint = {
            'epoch': epoch,
            'global_step': self.global_step,
            'model_state_dict': self.model.state_dict(),
            'optimizer_state_dict': optimizer.state_dict(),
            'scheduler_state_dict': scheduler.state_dict() if scheduler else None,
            'metrics': metrics,
            'stage': stage_name,
            'training_history': self.training_history,
            'config': self.config.__dict__
        }
        
        checkpoint_path = self.save_dir / f'best_{stage_name}.pth'
        torch.save(checkpoint, checkpoint_path)
        self.training_history['checkpoints'].append(str(checkpoint_path))
        
        print(f"💾 Best checkpoint saved: {checkpoint_path}")
    
    def save_stage_checkpoint(self, stage_name, best_metric):
        """Save stage completion checkpoint."""
        stage_checkpoint = {
            'stage': stage_name,
            'best_metric': best_metric,
            'model_state_dict': self.model.state_dict(),
            'training_history': self.training_history,
            'timestamp': time.time()
        }
        
        stage_path = self.save_dir / f'stage_{stage_name}_complete.pth'
        torch.save(stage_checkpoint, stage_path)
        
        print(f"🏁 Stage checkpoint saved: {stage_path}")

    def _initialize_gradcam_hooks(self):
        """Initialize Grad-CAM hooks on key model layers."""
        try:
            def save_activation(name):
                def hook(module, input, output):
                    self.gradcam_activations[name] = output.detach()
                return hook
            
            def save_gradient(name):
                def hook(module, input, grad_output):
                    if grad_output[0] is not None:
                        self.gradcam_gradients[name] = grad_output[0].detach()
                return hook
            
            # Register hooks on multiple layers for comprehensive analysis
            target_layers = []
            
            # Backbone layers
            if hasattr(self.model.backbone, 'rgb_backbone'):
                try:
                    # Try to access stages or blocks
                    if hasattr(self.model.backbone.rgb_backbone, 'stages'):
                        target_layers.append(('backbone_stage3', self.model.backbone.rgb_backbone.stages[3]))
                    elif hasattr(self.model.backbone.rgb_backbone, 'blocks'):
                        target_layers.append(('backbone_final', self.model.backbone.rgb_backbone.blocks[-1]))
                except:
                    target_layers.append(('backbone_final', self.model.backbone.rgb_backbone))
            
            # Weather classifier
            if hasattr(self.model.weather_classifier, 'encoder'):
                target_layers.append(('weather_encoder', self.model.weather_classifier.encoder[0]))
            
            # Register hooks
            for name, layer in target_layers:
                try:
                    if hasattr(layer, 'register_forward_hook'):
                        self.gradcam_hooks[f'{name}_forward'] = layer.register_forward_hook(save_activation(name))
                        self.gradcam_hooks[f'{name}_backward'] = layer.register_full_backward_hook(save_gradient(name))
                except Exception as e:
                    print(f"⚠️ Could not register hook for {name}: {e}")
                    continue
            
            print(f"🎯 Grad-CAM hooks registered: {len(self.gradcam_hooks)//2} layers")
            
        except Exception as e:
            print(f"⚠️ Grad-CAM hook initialization failed: {e}")

    def generate_comprehensive_explanation(self, image, weather_label=None, include_shap=True):
        """Generate comprehensive explanation for a single image."""
        explanations = {
            'gradcam': {},
            'shap': None,
            'attention': {},
            'feature_importance': {},
            'prediction_info': {},
            'saliency': {}
        }
        
        try:
            # Prepare image
            if isinstance(image, np.ndarray):
                image = torch.from_numpy(image).float()
            if len(image.shape) == 3:
                image = image.unsqueeze(0)
            image = image.to(self.device)
            
            # Get model prediction
            with torch.no_grad():
                outputs = self.model(image, None)
                weather_logits = outputs['weather_logits']
                weather_probs = F.softmax(weather_logits, dim=1)
                predicted_class = torch.argmax(weather_probs, dim=1).item()
                prediction_confidence = weather_probs[0, predicted_class].item()
            
            # Store prediction information
            explanations['prediction_info'] = {
                'predicted_class': predicted_class,
                'predicted_weather': WEATHER_NAMES[predicted_class],
                'confidence': prediction_confidence,
                'true_weather': WEATHER_NAMES[weather_label] if weather_label is not None else 'Unknown',
                'all_probabilities': weather_probs[0].cpu().numpy().tolist(),
                'prediction_entropy': -torch.sum(weather_probs * torch.log(weather_probs + 1e-8)).item()
            }
            
            # Generate Grad-CAM explanations
            explanations['gradcam'] = self._generate_gradcam_explanations(image, predicted_class)
            
            # Generate attention maps
            explanations['attention'] = self._generate_attention_maps(image, outputs)
            
            # Generate feature importance
            explanations['feature_importance'] = self._analyze_feature_importance(image, outputs)
            
            # Generate saliency maps
            explanations['saliency'] = self._generate_saliency_maps(image, predicted_class)
            
            # Generate SHAP if available and requested
            if self.shap_available and include_shap:
                explanations['shap'] = self._generate_shap_explanation(image)
            
            return explanations
            
        except Exception as e:
            print(f"⚠️ Comprehensive explanation generation failed: {e}")
            return explanations
    
    def _generate_gradcam_explanations(self, image, target_class):
        """Generate Grad-CAM heatmaps for all registered layers."""
        gradcam_maps = {}
        
        try:
            # Clear previous activations
            self.gradcam_activations.clear()
            self.gradcam_gradients.clear()
            
            # Enable gradients for input
            image.requires_grad_(True)
            
            # Forward pass
            outputs = self.model(image, None)
            weather_logits = outputs['weather_logits']
            
            # Backward pass for target class
            self.model.zero_grad()
            class_score = weather_logits[0, target_class]
            class_score.backward(retain_graph=True)
            
            # Generate heatmaps for each layer
            for layer_name in self.gradcam_activations.keys():
                if layer_name in self.gradcam_gradients:
                    try:
                        activation = self.gradcam_activations[layer_name]
                        gradient = self.gradcam_gradients[layer_name]
                        
                        # Global average pooling of gradients
                        weights = torch.mean(gradient, dim=[2, 3], keepdim=True)
                        
                        # Weighted combination of activation maps
                        weighted_activation = activation * weights
                        heatmap = torch.sum(weighted_activation, dim=1, keepdim=True)
                        
                        # Apply ReLU and normalize
                        heatmap = F.relu(heatmap)
                        if heatmap.max() > 0:
                            heatmap = heatmap / heatmap.max()
                        
                        # Resize to input size
                        heatmap_resized = F.interpolate(
                            heatmap,
                            size=(image.size(2), image.size(3)),
                            mode='bilinear',
                            align_corners=False
                        )
                        
                        gradcam_maps[layer_name] = heatmap_resized.squeeze().cpu().numpy()
                        
                    except Exception as e:
                        print(f"⚠️ Grad-CAM generation failed for {layer_name}: {e}")
                        continue
            
            return gradcam_maps
            
        except Exception as e:
            print(f"⚠️ Grad-CAM generation failed: {e}")
            return {}
    
    def _generate_attention_maps(self, image, outputs):
        """Generate comprehensive attention maps from model's internal mechanisms."""
        attention_maps = {}
        
        try:
            with torch.no_grad():
                # Get backbone features
                features = outputs.get('features', [])
                if not features:
                    features = self.model.backbone(image, None)
                
                # Weather gate attention
                for i, (feature, gate) in enumerate(zip(features, self.model.weather_gates)):
                    try:
                        gate_output = gate(feature)  # [B, num_weather_classes]
                        
                        # Convert to spatial attention
                        spatial_attention = gate_output.mean(dim=1, keepdim=True).unsqueeze(-1).unsqueeze(-1)
                        spatial_attention = F.interpolate(
                            spatial_attention,
                            size=(image.size(2), image.size(3)),
                            mode='bilinear',
                            align_corners=False
                        )
                        
                        attention_maps[f'weather_gate_{i}'] = spatial_attention.squeeze().cpu().numpy()
                        
                    except Exception as e:
                        print(f"⚠️ Weather gate attention failed for level {i}: {e}")
                        continue
                
                # Channel attention analysis
                for i, feature in enumerate(features):
                    try:
                        # Global average pooling for channel attention
                        channel_importance = torch.mean(feature, dim=[2, 3])  # [B, C]
                        channel_weights = F.softmax(channel_importance, dim=1)
                        
                        # Apply channel weights and create spatial map
                        weighted_feature = feature * channel_weights.unsqueeze(-1).unsqueeze(-1)
                        spatial_attention = torch.mean(weighted_feature, dim=1, keepdim=True)
                        
                        # Resize to input size
                        spatial_attention = F.interpolate(
                            spatial_attention,
                            size=(image.size(2), image.size(3)),
                            mode='bilinear',
                            align_corners=False
                        )
                        
                        attention_maps[f'channel_attention_{i}'] = spatial_attention.squeeze().cpu().numpy()
                        
                    except Exception as e:
                        print(f"⚠️ Channel attention failed for level {i}: {e}")
                        continue
                
                # Self-attention maps (if available in model)
                self._extract_self_attention_maps(features, attention_maps, image.size())
            
            return attention_maps
            
        except Exception as e:
            print(f"⚠️ Attention map generation failed: {e}")
            return {}
    
    def _extract_self_attention_maps(self, features, attention_maps, input_size):
        """Extract self-attention maps from transformer blocks if available."""
        try:
            for i, feature in enumerate(features):
                # Compute self-attention weights
                B, C, H, W = feature.shape
                
                # Reshape for attention computation
                feature_flat = feature.view(B, C, H * W).permute(0, 2, 1)  # [B, HW, C]
                
                # Compute attention weights (simplified)
                attention_weights = torch.bmm(feature_flat, feature_flat.transpose(1, 2))  # [B, HW, HW]
                attention_weights = F.softmax(attention_weights, dim=-1)
                
                # Average attention weights across spatial locations
                avg_attention = attention_weights.mean(dim=1)  # [B, HW]
                
                # Reshape back to spatial
                spatial_attention = avg_attention.view(B, 1, H, W)
                
                # Resize to input size
                spatial_attention = F.interpolate(
                    spatial_attention,
                    size=(input_size[2], input_size[3]),
                    mode='bilinear',
                    align_corners=False
                )
                
                attention_maps[f'self_attention_{i}'] = spatial_attention.squeeze().cpu().numpy()
                
        except Exception as e:
            print(f"⚠️ Self-attention extraction failed: {e}")
    
    def _analyze_feature_importance(self, image, outputs):
        """Analyze feature importance across different scales and channels."""
        feature_importance = {}
        
        try:
            features = outputs.get('features', [])
            if not features:
                features = self.model.backbone(image, None)
            
            for i, feature in enumerate(features):
                # Spatial importance (magnitude of activations)
                spatial_importance = torch.mean(torch.abs(feature), dim=1)  # [B, H, W]
                feature_importance[f'spatial_scale_{i}'] = spatial_importance.squeeze().cpu().numpy()
                
                # Channel importance (global average of each channel)
                channel_importance = torch.mean(torch.abs(feature), dim=[2, 3])  # [B, C]
                feature_importance[f'channel_scale_{i}'] = channel_importance.squeeze().cpu().numpy()
                
                # Feature variance (indicates information content)
                feature_variance = torch.var(feature, dim=[2, 3])  # [B, C]
                feature_importance[f'variance_scale_{i}'] = feature_variance.squeeze().cpu().numpy()
            
            return feature_importance
            
        except Exception as e:
            print(f"⚠️ Feature importance analysis failed: {e}")
            return {}
    
    def _generate_saliency_maps(self, image, target_class):
        """Generate vanilla gradient saliency maps."""
        saliency_maps = {}
        
        try:
            # Enable gradients for input
            image.requires_grad_(True)
            
            # Forward pass
            outputs = self.model(image, None)
            weather_logits = outputs['weather_logits']
            
            # Backward pass
            self.model.zero_grad()
            class_score = weather_logits[0, target_class]
            class_score.backward()
            
            # Get input gradients
            input_gradients = image.grad.data
            
            # Vanilla gradient saliency
            saliency = torch.abs(input_gradients).max(dim=1)[0]  # Max across RGB channels
            saliency_maps['vanilla_gradient'] = saliency.squeeze().cpu().numpy()
            
            # Integrated gradients (simplified version)
            saliency_maps['integrated_gradients'] = self._compute_integrated_gradients(image, target_class)
            
            return saliency_maps
            
        except Exception as e:
            print(f"⚠️ Saliency map generation failed: {e}")
            return {}
    
    def _compute_integrated_gradients(self, image, target_class, steps=20):
        """Compute simplified integrated gradients."""
        try:
            # Create baseline (black image)
            baseline = torch.zeros_like(image)
            
            # Create interpolated images
            integrated_grad = torch.zeros_like(image)
            
            for i in range(steps):
                alpha = i / steps
                interpolated = baseline + alpha * (image - baseline)
                interpolated.requires_grad_(True)
                
                # Forward pass
                outputs = self.model(interpolated, None)
                weather_logits = outputs['weather_logits']
                
                # Backward pass
                self.model.zero_grad()
                class_score = weather_logits[0, target_class]
                class_score.backward()
                
                # Accumulate gradients
                integrated_grad += interpolated.grad.data
            
            # Average and multiply by input difference
            integrated_grad = integrated_grad / steps
            integrated_grad = integrated_grad * (image - baseline)
            
            # Return magnitude across channels
            return torch.abs(integrated_grad).max(dim=1)[0].squeeze().cpu().numpy()
            
        except Exception as e:
            print(f"⚠️ Integrated gradients computation failed: {e}")
            return np.zeros((512, 512))
    
    def _generate_shap_explanation(self, image):
        """Generate SHAP explanations if available."""
        if not self.shap_available:
            return None
        
        try:
            # Convert to numpy for SHAP
            image_np = image.squeeze().cpu().numpy()
            
            # Generate SHAP values
            shap_values = self.shap_explainer.shap_values(image_np, max_evals=100)
            
            return {
                'shap_values': shap_values,
                'expected_value': self.shap_explainer.expected_value,
                'feature_attribution': np.sum(np.abs(shap_values), axis=0) if isinstance(shap_values, list) else np.abs(shap_values)
            }
            
        except Exception as e:
            print(f"⚠️ SHAP explanation generation failed: {e}")
            return None

    def estimate_comprehensive_uncertainty(self, dataloader, max_batches=20):
        """Estimate comprehensive uncertainty using multiple methods."""
        print("🔍 Estimating comprehensive uncertainty...")
        
        all_uncertainties = []
        all_metadata = []
        
        # Enable dropout for uncertainty estimation
        self.model.train()
        
        with torch.no_grad():
            for batch_idx, batch in enumerate(tqdm(dataloader, desc="Uncertainty estimation")):

                    uncertainties = self._compute_multi_method_uncertainty(rgb, nir)
                    
                    # Process each sample in the batch
                    for i in range(rgb.size(0)):

                        }

        self.model.eval()

        return all_uncertainties, all_metadata
    
    def _compute_multi_method_uncertainty(self, rgb, nir):
        """Compute uncertainty using multiple methods."""
        uncertainties = {}
        
        # MC Dropout uncertainty
        uncertainties['mc_dropout'] = self._mc_dropout_uncertainty(rgb, nir)
        
        # Predictive entropy
        uncertainties['predictive_entropy'] = self._predictive_entropy(rgb, nir)
        
        # BALD (Bayesian Active Learning by Disagreement)
        uncertainties['bald'] = self._bald_uncertainty(rgb, nir)
        
        # Ensemble disagreement (simplified)
        uncertainties['ensemble_disagreement'] = self._ensemble_disagreement(rgb, nir)
        
        return uncertainties
    
    def _mc_dropout_uncertainty(self, rgb, nir):
        """Estimate uncertainty using Monte Carlo Dropout."""
        predictions = []
        
        for _ in range(self.num_mc_samples):
            outputs = self.model(rgb, nir)
            weather_probs = F.softmax(outputs['weather_logits'], dim=1)
            predictions.append(weather_probs)
        
        predictions = torch.stack(predictions, dim=0)  # [num_samples, batch_size, num_classes]
        
        # Compute predictive entropy
        mean_pred = predictions.mean(dim=0)
        predictive_entropy = -torch.sum(mean_pred * torch.log(mean_pred + 1e-8), dim=1)
        
        return predictive_entropy.cpu().numpy()
    
    def _predictive_entropy(self, rgb, nir):
        """Compute predictive entropy uncertainty."""
        outputs = self.model(rgb, nir)
        weather_probs = F.softmax(outputs['weather_logits'], dim=1)
        
        entropy = -torch.sum(weather_probs * torch.log(weather_probs + 1e-8), dim=1)
        return entropy.cpu().numpy()
    
    def _bald_uncertainty(self, rgb, nir):
        """Compute BALD (Bayesian Active Learning by Disagreement) uncertainty."""
        predictions = []
        
        for _ in range(self.num_mc_samples):
            outputs = self.model(rgb, nir)
            weather_probs = F.softmax(outputs['weather_logits'], dim=1)
            predictions.append(weather_probs)
        
        predictions = torch.stack(predictions, dim=0)
        
        # Mutual information (epistemic uncertainty)
        mean_pred = predictions.mean(dim=0)
        predictive_entropy = -torch.sum(mean_pred * torch.log(mean_pred + 1e-8), dim=1)
        
        individual_entropies = -torch.sum(predictions * torch.log(predictions + 1e-8), dim=2)
        expected_entropy = individual_entropies.mean(dim=0)
        
        mutual_info = predictive_entropy - expected_entropy
        return mutual_info.cpu().numpy()
    
    def _ensemble_disagreement(self, rgb, nir):
        """Compute ensemble disagreement using different dropout patterns."""
        predictions = []
        
        # Use different random seeds for pseudo-ensemble
        for i in range(5):
            torch.manual_seed(42 + i)
            outputs = self.model(rgb, nir)
            weather_probs = F.softmax(outputs['weather_logits'], dim=1)
            predictions.append(weather_probs)
        
        predictions = torch.stack(predictions, dim=0)
        disagreement = predictions.var(dim=0).mean(dim=1)
        
        return disagreement.cpu().numpy()
    
    def _compute_composite_uncertainty(self, sample_uncertainties):
        """Compute composite uncertainty score from multiple methods."""
        weights = {
            'mc_dropout': 0.3,
            'predictive_entropy': 0.2,
            'bald': 0.3,
            'ensemble_disagreement': 0.2
        }
        
        composite = 0.0
        total_weight = 0.0
        
        for method, uncertainty in sample_uncertainties.items():
            if method in weights:
                composite += weights[method] * uncertainty
                total_weight += weights[method]
        
        return composite / total_weight if total_weight > 0 else 0.0
    
    def select_high_uncertainty_samples(self, uncertainties, metadata, top_k=50):
        """Select high uncertainty samples with diversity constraints."""
        try:
            # Combine uncertainties with metadata
            combined_data = list(zip(uncertainties, metadata))
            
            # Sort by uncertainty (descending)
            combined_data.sort(key=lambda x: x[0], reverse=True)
            
            # Advanced selection with diversity constraints
            selected_samples = []
            weather_counts = defaultdict(int)
            source_counts = defaultdict(int)
            
            # Limits for diversity
            max_per_weather = max(2, top_k // len(WEATHER_NAMES))
            max_per_source = max(5, top_k // 3)
            
            for uncertainty, meta in combined_data:
                if len(selected_samples) >= top_k:
                    break
                
                weather = meta['weather_name']
                source = meta['source_dataset']
                
                # Check diversity constraints
                weather_ok = weather_counts[weather] < max_per_weather
                source_ok = source_counts[source] < max_per_source
                uncertainty_ok = uncertainty > self.uncertainty_threshold
                
                if weather_ok and source_ok and uncertainty_ok:
                    selected_samples.append({
                        **meta,
                        'selection_reason': 'high_uncertainty',
                        'selection_rank': len(selected_samples) + 1,
                        'uncertainty_score': uncertainty
                    })
                    
                    weather_counts[weather] += 1
                    source_counts[source] += 1
            
            # If we don't have enough, relax constraints
            if len(selected_samples) < top_k // 2:
                for uncertainty, meta in combined_data:
                    if len(selected_samples) >= top_k:
                        break
                    
                    # Check if already selected
                    already_selected = any(
                        s['image_id'] == meta['image_id'] for s in selected_samples
                    )
                    
                    if not already_selected and uncertainty > self.uncertainty_threshold * 0.7:
                        selected_samples.append({
                            **meta,
                            'selection_reason': 'high_uncertainty_relaxed',
                            'selection_rank': len(selected_samples) + 1,
                            'uncertainty_score': uncertainty
                        })
            
            self.selection_history['uncertain_samples'].extend(selected_samples)
            
            print(f"🎯 Selected {len(selected_samples)} high-uncertainty samples")
            print(f"   Weather distribution: {dict(weather_counts)}")
            print(f"   Source distribution: {dict(source_counts)}")
            
            return selected_samples
            
        except Exception as e:
            print(f"⚠️ High uncertainty sample selection failed: {e}")
            return []

            feature_matrix = torch.cat(all_features, dim=0).numpy()
            
            # Advanced clustering for diversity
            diverse_samples = self._perform_advanced_clustering(

    def _perform_advanced_clustering(self, features, metadata, top_k):
        """Perform advanced clustering for diversity selection."""
        try:
            diverse_samples = []
            
            # Method 1: K-means clustering
            n_clusters = min(top_k, len(features))
            kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
            cluster_labels = kmeans.fit_predict(features)
            
            # Select representative from each cluster
            for cluster_id in range(n_clusters):
                cluster_mask = (cluster_labels == cluster_id)
                if not cluster_mask.any():
                    continue
                
                cluster_indices = np.where(cluster_mask)[0]
                cluster_features = features[cluster_mask]
                
                # Find sample closest to cluster center
                center = kmeans.cluster_centers_[cluster_id]
                distances = pairwise_distances([center], cluster_features)[0]
                closest_idx = np.argmin(distances)
                
                original_idx = cluster_indices[closest_idx]
                if original_idx < len(metadata):
                    diverse_samples.append({
                        **metadata[original_idx],
                        'selection_reason': 'diversity_kmeans',
                        'cluster_id': cluster_id,
                        'distance_to_center': distances[closest_idx],
                        'diversity_score': 1.0 / (distances[closest_idx] + 1e-6)
                    })
            
            # Method 2: Maximum distance selection
            if len(diverse_samples) < top_k:
                remaining_needed = top_k - len(diverse_samples)
                selected_indices = set(
                    s.get('batch_idx', 0) * 1000 + s.get('sample_idx', 0)
                    for s in diverse_samples
                )
                
                for _ in range(min(remaining_needed, len(metadata) - len(diverse_samples))):
                    max_min_distance = -1
                    best_candidate_idx = -1
                    
                    for candidate_idx in range(len(metadata)):
                        candidate_key = (metadata[candidate_idx].get('batch_idx', 0) * 1000 +
                                       metadata[candidate_idx].get('sample_idx', 0))
                        
                        if candidate_key in selected_indices:
                            continue
                        
                        # Calculate minimum distance to selected samples
                        if len(diverse_samples) == 0:
                            min_distance = float('inf')
                        else:
                            min_distance = float('inf')
                            candidate_features = features[candidate_idx % len(features)]
                            
                            for selected_sample in diverse_samples:
                                try:
                                    selected_idx = (selected_sample.get('batch_idx', 0) * 1000 +
                                                  selected_sample.get('sample_idx', 0)) % len(features)
                                    selected_features = features[selected_idx]
                                    distance = np.linalg.norm(candidate_features - selected_features)
                                    min_distance = min(min_distance, distance)
                                except:
                                    continue
                        
                        if min_distance > max_min_distance:
                            max_min_distance = min_distance
                            best_candidate_idx = candidate_idx
                    
                    if best_candidate_idx != -1:
                        diverse_samples.append({
                            **metadata[best_candidate_idx],
                            'selection_reason': 'max_distance',
                            'min_distance_to_selected': max_min_distance
                        })
                        
                        candidate_key = (metadata[best_candidate_idx].get('batch_idx', 0) * 1000 +
                                       metadata[best_candidate_idx].get('sample_idx', 0))
                        selected_indices.add(candidate_key)
            
            return diverse_samples[:top_k]
            
        except Exception as e:
            print(f"⚠️ Advanced clustering failed: {e}")
            return []

    def _create_policy_network(self):
        """Create advanced policy network with residual connections and layer normalization."""
        class PolicyNetwork(nn.Module):
            def __init__(self, state_dim, action_dim, hidden_dim):
                super().__init__()
                self.state_dim = state_dim
                self.action_dim = action_dim
                
                # Encoder layers
                self.encoder = nn.Sequential(
                    nn.Linear(state_dim, hidden_dim),
                    nn.LayerNorm(hidden_dim),
                    nn.ReLU(inplace=True),
                    nn.Dropout(0.2)
                )
                
                # Residual blocks
                self.residual1 = self._create_residual_block(hidden_dim)
                self.residual2 = self._create_residual_block(hidden_dim)
                
                # Output heads
                self.mean_head = nn.Sequential(
                    nn.Linear(hidden_dim, hidden_dim // 2),
                    nn.LayerNorm(hidden_dim // 2),
                    nn.ReLU(inplace=True),
                    nn.Linear(hidden_dim // 2, action_dim),
                    nn.Tanh()  # Bounded output
                )
                
                self.std_head = nn.Sequential(
                    nn.Linear(hidden_dim, hidden_dim // 2),
                    nn.LayerNorm(hidden_dim // 2),
                    nn.ReLU(inplace=True),
                    nn.Linear(hidden_dim // 2, action_dim),
                    nn.Softplus()  # Positive output
                )
                
                # Initialize weights
                self.apply(self._init_weights)
            
            def _create_residual_block(self, dim):
                return nn.Sequential(
                    nn.Linear(dim, dim),
                    nn.LayerNorm(dim),
                    nn.ReLU(inplace=True),
                    nn.Dropout(0.1),
                    nn.Linear(dim, dim),
                    nn.LayerNorm(dim)
                )
            
            def _init_weights(self, module):
                if isinstance(module, nn.Linear):
                    torch.nn.init.orthogonal_(module.weight, gain=0.01)
                    torch.nn.init.constant_(module.bias, 0)
            
            def forward(self, state):
                x = self.encoder(state)
                
                # Residual connections
                residual = x
                x = self.residual1(x)
                x = F.relu(x + residual)
                
                residual = x
                x = self.residual2(x)
                x = F.relu(x + residual)
                
                # Output heads
                mean = self.mean_head(x)
                std = self.std_head(x) + 1e-6  # Stability
                
                return mean, std
        
        return PolicyNetwork(self.state_dim, self.action_dim, self.hidden_dim)
    
    def _create_value_network(self):
        """Create value network for advantage estimation."""
        class ValueNetwork(nn.Module):
            def __init__(self, state_dim, hidden_dim):
                super().__init__()
                
                self.network = nn.Sequential(
                    nn.Linear(state_dim, hidden_dim),
                    nn.LayerNorm(hidden_dim),
                    nn.ReLU(inplace=True),
                    nn.Dropout(0.2),
                    
                    nn.Linear(hidden_dim, hidden_dim),
                    nn.LayerNorm(hidden_dim),
                    nn.ReLU(inplace=True),
                    nn.Dropout(0.1),
                    
                    nn.Linear(hidden_dim, hidden_dim // 2),
                    nn.LayerNorm(hidden_dim // 2),
                    nn.ReLU(inplace=True),
                    
                    nn.Linear(hidden_dim // 2, 1)
                )
                
                # Initialize weights
                self.apply(self._init_weights)
            
            def _init_weights(self, module):
                if isinstance(module, nn.Linear):
                    torch.nn.init.orthogonal_(module.weight, gain=1.0)
                    torch.nn.init.constant_(module.bias, 0)
            
            def forward(self, state):
                return self.network(state).squeeze(-1)
        
        return ValueNetwork(self.state_dim, self.hidden_dim)

        except Exception as e:
            print(f"⚠️ State extraction failed: {e}")
            # Return safe neutral state
            return torch.ones(self.state_dim, dtype=torch.float32).to(device) * 0.5
    
    def get_policy_action(self, state, deterministic=False):
        """Get action from policy with comprehensive action interpretation and logging."""
        try:
            with torch.no_grad():
                mean, std = self.policy_net(state.unsqueeze(0))
                mean = mean.squeeze(0)
                std = std.squeeze(0)
                
                if deterministic:
                    action_raw = mean
                    log_prob = None
                else:
                    # Create distribution and sample
                    dist_obj = torch.distributions.Normal(mean, std)
                    action_raw = dist_obj.sample()
                    log_prob = dist_obj.log_prob(action_raw).sum()
                
                # Convert to valid action space with semantic interpretation
                semantic_action = self._convert_to_semantic_actions(action_raw)
                
                # Store action distribution for analysis
                self.performance_metrics['action_distributions'].append(semantic_action.cpu().numpy())
                
                return semantic_action, log_prob, mean, std
                
        except Exception as e:
            print(f"⚠️ Policy action generation failed: {e}")
            # Return safe default actions
            default_action = torch.tensor([0.7, 0.8, 0.6, 0.7, 0.2, 0.8]).to(device)
            return default_action, None, None, None
    
    def _convert_to_semantic_actions(self, normalized_actions):
        """Convert normalized actions to semantic action space with safety constraints."""
        semantic_actions = []
        
        for i, norm_val in enumerate(normalized_actions):
            action_info = self.action_semantics[i]
            low, high = action_info['range']
            
            # Convert from [-1, 1] to [low, high] (tanh output range)
            semantic_val = low + (norm_val + 1) * (high - low) / 2
            
            # Apply safety constraints
            semantic_val = max(low, min(high, semantic_val.item()))
            semantic_actions.append(semantic_val)
        
        return torch.tensor(semantic_actions).to(device)
    
    def compute_comprehensive_reward(self, model_outputs, actions, ground_truth, system_metrics,

            uncertainty_handling = 1.0 - min(system_metrics.get('prediction_uncertainty', 0.5), 1.0)
            adversarial_robustness = system_metrics.get('adversarial_robustness', 0.7)
            safety_compliance = self._compute_safety_compliance(actions)

            xai_integration_reward = min(xai_quality, 1.0) * 0.6
            al_integration_reward = min(al_quality, 1.0) * 0.4
            integration_reward = (xai_integration_reward + al_integration_reward) * 0.15

            if safety_violations:

                total_reward -= violation_penalty

            action_penalty = self._compute_action_penalty(actions)

        except Exception as e:

    def _compute_weather_accuracy(self, outputs, ground_truth):
        """Compute weather classification accuracy with confidence weighting."""
        try:
            if 'weather_logits' in outputs and 'weather_label' in ground_truth:
                predictions = torch.argmax(outputs['weather_logits'], dim=1)
                targets = ground_truth['weather_label']
                
                # Basic accuracy
                accuracy = (predictions == targets).float().mean().item()
                
                # Confidence weighting
                probs = F.softmax(outputs['weather_logits'], dim=1)
                confidence = torch.max(probs, dim=1)[0].mean().item()
                
                # Weighted accuracy (higher confidence should correlate with higher accuracy)
                weighted_accuracy = accuracy * (0.6 + 0.4 * confidence)
                return min(weighted_accuracy, 1.0)
        except:
            pass
        return 0.5
    
    def _compute_confidence_calibration(self, outputs, ground_truth):
        """Assess how well model confidence correlates with actual correctness."""
        try:
            if 'weather_logits' in outputs and 'weather_label' in ground_truth:
                predictions = torch.argmax(outputs['weather_logits'], dim=1)
                targets = ground_truth['weather_label']
                probs = F.softmax(outputs['weather_logits'], dim=1)
                
                confidences = torch.max(probs, dim=1)[0]
                correctness = (predictions == targets).float()
                
                if len(confidences) > 1:
                    # Compute correlation between confidence and correctness
                    correlation = torch.corrcoef(torch.stack([confidences, correctness]))[0, 1]
                    # Convert from [-1, 1] to [0, 1]
                    calibration_quality = (correlation + 1) / 2
                    return calibration_quality.item() if not torch.isnan(calibration_quality) else 0.5
        except:
            pass
        return 0.5
    
    def _compute_segmentation_quality(self, outputs):
        """Compute segmentation quality based on spatial consistency and confidence."""
        try:
            if 'segmentation' in outputs:
                seg_probs = F.softmax(outputs['segmentation'], dim=1)
                
                # Confidence quality
                seg_confidence = torch.max(seg_probs, dim=1)[0].mean().item()
                
                # Spatial consistency (simplified)
                seg_preds = torch.argmax(seg_probs, dim=1)
                spatial_consistency = self._compute_spatial_consistency(seg_preds)
                
                return (seg_confidence * 0.6 + spatial_consistency * 0.4)
        except:
            pass
        return 0.5
    
    def _compute_spatial_consistency(self, segmentation_mask):
        """Compute spatial consistency of segmentation predictions."""
        try:
            if segmentation_mask.dim() > 2:
                mask = segmentation_mask[0].cpu().numpy()
            else:
                mask = segmentation_mask.cpu().numpy()
            
            # Compute gradient magnitude (lower = more consistent)
            if mask.shape[0] > 1 and mask.shape[1] > 1:
                grad_x = np.abs(np.diff(mask, axis=1))
                grad_y = np.abs(np.diff(mask, axis=0))
                
                max_val = mask.max()
                if max_val > 0:
                    consistency = 1.0 - (np.mean(grad_x) + np.mean(grad_y)) / (2 * max_val)
                    return max(0.0, min(1.0, consistency))
        except:
            pass
        return 0.5
    
    def _compute_safety_compliance(self, actions):
        """Assess safety compliance of action combinations."""
        try:
            detection_threshold = actions[0].item()
            processing_intensity = actions[1].item()
            uncertainty_handling = actions[3].item()
            system_conservatism = actions[5].item()
            
            safety_score = 1.0
            
            # Safety Rule 1: High detection threshold requires high conservatism
            if detection_threshold > 0.8 and system_conservatism < 0.5:
                safety_score *= 0.8
            
            # Safety Rule 2: Low uncertainty handling requires high conservatism
            if uncertainty_handling < 0.4 and system_conservatism < 0.6:
                safety_score *= 0.7
            
            # Safety Rule 3: Balanced approach is preferred
            action_balance = 1.0 - np.std([detection_threshold, uncertainty_handling, system_conservatism])
            safety_score *= (0.8 + 0.2 * action_balance)
            
            # Safety Rule 4: Extreme values are penalized
            extreme_penalty = sum([1 for a in actions if a.item() < 0.1 or a.item() > 0.95]) * 0.1
            safety_score -= extreme_penalty
            
            return max(0.0, min(1.0, safety_score))
        except:
            return 0.5
    
    def _compute_action_penalty(self, actions):
        """Compute penalty for unreasonable action combinations."""
        try:
            # Penalty for extreme values
            extreme_penalty = sum([0.05 for a in actions if a.item() < 0.1 or a.item() > 0.95])
            
            # Penalty for rapid changes (if we had history)
            # For now, just penalize extreme variance
            action_variance = torch.var(actions).item()
            variance_penalty = max(0, action_variance - 0.1) * 0.2
            
            return extreme_penalty + variance_penalty
        except:
            return 0.0
    
    def update_policy_ppo(self, experiences):
        """Advanced PPO update with comprehensive improvements and monitoring."""
        if len(experiences) < self.ppo_config['batch_size']:
            return {}
        
        try:
            # Prepare batch data
            states = torch.stack([exp['state'] for exp in experiences])
            actions = torch.stack([exp['action'] for exp in experiences])
            rewards = torch.stack([exp['reward'] for exp in experiences])
            old_log_probs = torch.stack([exp['log_prob'] for exp in experiences if exp.get('log_prob') is not None])
            dones = torch.tensor([exp.get('done', False) for exp in experiences], dtype=torch.bool).to(device)
            
            # Skip if we don't have log probs (deterministic actions)
            if len(old_log_probs) != len(experiences):
                print("⚠️ Skipping PPO update - missing log probabilities")
                return {}
            
            # Compute advantages using GAE
            advantages, returns = self._compute_gae_advantages(states, rewards, dones)
            
            # Normalize advantages
            if advantages.std() > 1e-8:
                advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)
            
            # Convert to dataset for batch processing
            dataset = torch.utils.data.TensorDataset(states, actions, returns, advantages, old_log_probs)
            dataloader = torch.utils.data.DataLoader(dataset, batch_size=self.ppo_config['batch_size'], shuffle=True)
            
            # Training metrics
            total_policy_loss = 0
            total_value_loss = 0
            total_entropy_loss = 0
            total_kl_div = 0
            clip_fraction = 0
            num_updates = 0
            
            # PPO training epochs
            for epoch in range(self.ppo_config['ppo_epochs']):
                for batch_states, batch_actions, batch_returns, batch_advantages, batch_old_log_probs in dataloader:
                    
                    # Policy forward pass
                    mean, std = self.policy_net(batch_states)
                    dist_obj = torch.distributions.Normal(mean, std)
                    new_log_probs = dist_obj.log_prob(batch_actions).sum(dim=-1)
                    entropy = dist_obj.entropy().sum(dim=-1)
                    
                    # Value forward pass
                    values = self.value_net(batch_states)
                    
                    # Compute policy loss (PPO clipped objective)
                    ratio = torch.exp(new_log_probs - batch_old_log_probs)
                    surr1 = ratio * batch_advantages
                    surr2 = torch.clamp(ratio, 1 - self.ppo_config['clip_ratio'], 1 + self.ppo_config['clip_ratio']) * batch_advantages
                    
                    policy_loss = -torch.min(surr1, surr2).mean()
                    
                    # Compute value loss
                    value_loss = F.mse_loss(values, batch_returns)
                    
                    # Compute entropy loss
                    entropy_loss = -entropy.mean()
                    
                    # Compute KL divergence for early stopping
                    with torch.no_grad():
                        kl_div = (batch_old_log_probs - new_log_probs).mean()
                        
                    # Combined loss
                    total_loss = (policy_loss + 
                                self.ppo_config['value_coef'] * value_loss + 
                                self.ppo_config['entropy_coef'] * entropy_loss)
                    
                    # Policy update
                    self.policy_optimizer.zero_grad()
                    self.value_optimizer.zero_grad()
                    total_loss.backward()
                    
                    # Gradient clipping
                    policy_grad_norm = torch.nn.utils.clip_grad_norm_(self.policy_net.parameters(), self.ppo_config['max_grad_norm'])
                    value_grad_norm = torch.nn.utils.clip_grad_norm_(self.value_net.parameters(), self.ppo_config['max_grad_norm'])
                    
                    self.policy_optimizer.step()
                    self.value_optimizer.step()
                    
                    # Accumulate metrics
                    total_policy_loss += policy_loss.item()
                    total_value_loss += value_loss.item()
                    total_entropy_loss += entropy_loss.item()
                    total_kl_div += kl_div.item()
                    
                    # Clip fraction (how often we clipped)
                    clipped = ((ratio < 1 - self.ppo_config['clip_ratio']) | 
                             (ratio > 1 + self.ppo_config['clip_ratio'])).float()
                    clip_fraction += clipped.mean().item()
                    
                    num_updates += 1
                    
                    # Store gradient norms
                    self.performance_metrics['gradient_norms'].append((policy_grad_norm.item() + value_grad_norm.item()) / 2)
                    
                    # Early stopping based on KL divergence
                    if kl_div > self.ppo_config['target_kl'] * 1.5:
                        print(f"Early stopping due to high KL divergence: {kl_div:.4f}")
                        break
                
                # Early stopping check
                if total_kl_div / max(num_updates, 1) > self.ppo_config['target_kl'] * 1.5:
                    break
            
            # Average metrics
            avg_policy_loss = total_policy_loss / max(num_updates, 1)
            avg_value_loss = total_value_loss / max(num_updates, 1)
            avg_entropy_loss = total_entropy_loss / max(num_updates, 1)
            avg_kl_div = total_kl_div / max(num_updates, 1)
            avg_clip_fraction = clip_fraction / max(num_updates, 1)
            
            # Update performance tracking
            avg_reward = rewards.mean().item()
            self.performance_metrics['rewards'].append(avg_reward)
            self.performance_metrics['policy_losses'].append(avg_policy_loss)
            self.performance_metrics['value_losses'].append(avg_value_loss)
            self.performance_metrics['entropy_losses'].append(avg_entropy_loss)
            self.performance_metrics['kl_divergences'].append(avg_kl_div)
            
            # Compute explained variance
            explained_var = 1 - torch.var(returns - values.detach()) / torch.var(returns)
            self.performance_metrics['explained_variance'].append(explained_var.item())
            
            # Update training statistics
            self.training_stats['total_steps'] += len(experiences)
            self.training_stats['successful_episodes'] += sum(1 for r in rewards if r > 0)
            
            if avg_reward > self.training_stats['best_reward']:
                self.training_stats['best_reward'] = avg_reward
            
            # Compute convergence and stability scores
            self._update_convergence_metrics()
            
            return {
                'policy_loss': avg_policy_loss,
                'value_loss': avg_value_loss,
                'entropy_loss': avg_entropy_loss,
                'kl_divergence': avg_kl_div,
                'clip_fraction': avg_clip_fraction,
                'mean_reward': avg_reward,
                'std_reward': rewards.std().item(),
                'explained_variance': explained_var.item(),
                'num_updates': num_updates,
                'gradient_norm': self.performance_metrics['gradient_norms'][-1] if self.performance_metrics['gradient_norms'] else 0.0
            }
            
        except Exception as e:
            print(f"⚠️ PPO update failed: {e}")
            import traceback
            traceback.print_exc()
            return {}
    
    def _compute_gae_advantages(self, states, rewards, dones):
        """Compute Generalized Advantage Estimation (GAE) advantages."""
        try:
            # Get value estimates
            with torch.no_grad():
                values = self.value_net(states)
            
            advantages = torch.zeros_like(rewards)
            returns = torch.zeros_like(rewards)
            
            gae = 0
            next_value = 0
            
            for t in reversed(range(len(rewards))):
                if t == len(rewards) - 1:
                    next_non_terminal = 1.0 - dones[t].float()
                    next_value = 0
                else:
                    next_non_terminal = 1.0 - dones[t].float()
                    next_value = values[t + 1]
                
                delta = rewards[t] + self.ppo_config['gamma'] * next_value * next_non_terminal - values[t]
                gae = delta + self.ppo_config['gamma'] * self.ppo_config['gae_lambda'] * next_non_terminal * gae
                advantages[t] = gae
                returns[t] = gae + values[t]
            
            return advantages, returns
            
        except Exception as e:
            print(f"⚠️ GAE computation failed: {e}")
            # Fallback to simple advantage computation
            returns = rewards.clone()
            values = torch.zeros_like(rewards)
            advantages = returns - values
            return advantages, returns
    
    def _update_convergence_metrics(self):
        """Update convergence and stability metrics."""
        try:
            if len(self.performance_metrics['rewards']) >= 100:
                recent_rewards = list(self.performance_metrics['rewards'])[-100:]
                
                # Convergence score (based on reward trend)
                if len(recent_rewards) > 50:
                    first_half = np.mean(recent_rewards[:50])
                    second_half = np.mean(recent_rewards[50:])
                    improvement = (second_half - first_half) / (abs(first_half) + 1e-8)
                    self.training_stats['convergence_score'] = max(0, min(1, improvement + 0.5))
                
                # Stability score (based on reward variance)
                reward_std = np.std(recent_rewards)
                reward_mean = np.mean(recent_rewards)
                cv = reward_std / (abs(reward_mean) + 1e-8)  # Coefficient of variation
                self.training_stats['stability_score'] = max(0, 1 - cv)
        
        except Exception as e:
            print(f"⚠️ Convergence metrics update failed: {e}")
    
    def get_training_summary(self):
        """Get comprehensive training summary."""
        try:
            summary = {
                'training_stats': self.training_stats.copy(),
                'recent_performance': {
                    'mean_reward_last_100': np.mean(list(self.performance_metrics['rewards'])[-100:]) if len(self.performance_metrics['rewards']) >= 100 else 0,
                    'mean_policy_loss_last_50': np.mean(list(self.performance_metrics['policy_losses'])[-50:]) if len(self.performance_metrics['policy_losses']) >= 50 else 0,
                    'mean_value_loss_last_50': np.mean(list(self.performance_metrics['value_losses'])[-50:]) if len(self.performance_metrics['value_losses']) >= 50 else 0,
                    'mean_kl_div_last_50': np.mean(list(self.performance_metrics['kl_divergences'])[-50:]) if len(self.performance_metrics['kl_divergences']) >= 50 else 0
                },
                'action_analysis': {
                    'mean_actions': np.mean(list(self.performance_metrics['action_distributions'])[-20:], axis=0).tolist() if len(self.performance_metrics['action_distributions']) >= 20 else [0.5] * self.action_dim,
                    'action_std': np.std(list(self.performance_metrics['action_distributions'])[-20:], axis=0).tolist() if len(self.performance_metrics['action_distributions']) >= 20 else [0.1] * self.action_dim
                }
            }
            
            return summary
        except Exception as e:
            print(f"⚠️ Training summary generation failed: {e}")
            return {'error': str(e)}

class RLTrainingCoordinator:
    """Coordinates RL training with the main model and other components."""
    
    def __init__(self, model, rl_policy, xai_explainer=None, active_learner=None):
        self.model = model
        self.rl_policy = rl_policy
        self.xai_explainer = xai_explainer
        self.active_learner = active_learner
        
        # Training configuration
        self.training_config = {
            'episodes_per_batch': 32,
            'max_episode_length': 100,
            'update_frequency': 5,
            'save_frequency': 50,
            'evaluation_frequency': 25
        }
        
        # Episode tracking
        self.current_episode = 0
        self.episodes_data = []
        
        print("🎮 RL Training Coordinator initialized")
    
    def run_rl_training_episode(self, dataloader, episode_length=50):
        """Run a single RL training episode."""
        episode_experiences = []
        episode_reward = 0
        
        try:
            self.model.eval()
            
            for step, batch in enumerate(dataloader):
                if step >= episode_length:
                    break
                
                # Prepare batch
                rgb = batch['rgb'].to(device)
                nir = batch.get('nir')
                if nir is not None and nir[0] is not None:
                    nir = torch.stack([n if n is not None else torch.zeros_like(rgb[0:1]) for n in nir]).to(device)
                
                targets = {'weather_label': batch['weather_label'].to(device)}
                
                # Model forward pass
                with torch.no_grad():
                    start_time = time.time()
                    outputs = self.model(rgb, nir)
                    inference_time = time.time() - start_time
                
                # Extract system metrics
                system_metrics = {
                    'accuracy_estimate': (torch.argmax(outputs['weather_logits'], dim=1) == targets['weather_label']).float().mean().item(),
                    'processing_time': inference_time,
                    'memory_usage': torch.cuda.memory_allocated() / (1024**3) if torch.cuda.is_available() else 0.5,
                    'gpu_utilization': torch.cuda.memory_allocated() / torch.cuda.max_memory_allocated() if torch.cuda.is_available() else 0.5,
                    'prediction_uncertainty': (1.0 - torch.max(F.softmax(outputs['weather_logits'], dim=1), dim=1)[0]).mean().item()
                }
                
                batch_info = {
                    'weather_types': batch['weather_name'],
                    'difficulty_score': np.random.uniform(0.3, 0.8),  # Simulated
                    'synthetic_ratio': sum(batch['is_synthetic']) / len(batch['is_synthetic'])
                }
                
                # Extract RL state
                state = self.rl_policy.extract_comprehensive_state(outputs, system_metrics, batch_info)
                
                # Get RL action
                action, log_prob, action_mean, action_std = self.rl_policy.get_policy_action(state, deterministic=False)
                
                # Compute reward
                xai_quality = 0.8 if self.xai_explainer else 0.5
                al_quality = 0.7 if self.active_learner else 0.5
                
                reward = self.rl_policy.compute_comprehensive_reward(
                    outputs, action, targets, system_metrics, xai_quality, al_quality
                )
                
                # Store experience
                if log_prob is not None:
                    experience = {
                        'state': state,
                        'action': action,
                        'reward': reward,
                        'log_prob': log_prob,
                        'done': (step == episode_length - 1)
                    }
                    episode_experiences.append(experience)
                
                episode_reward += reward.item()
            
            # Store episode data
            self.episodes_data.append({
                'episode': self.current_episode,
                'total_reward': episode_reward,
                'episode_length': len(episode_experiences),
                'average_reward': episode_reward / max(len(episode_experiences), 1)
            })
            
            self.current_episode += 1
            
            return episode_experiences, episode_reward
            
        except Exception as e:
            print(f"⚠️ RL training episode failed: {e}")
            return [], 0.0
    
    def run_comprehensive_rl_training(self, dataloader, num_episodes=100):
        """Run comprehensive RL training with monitoring and evaluation."""
        print(f"🚀 Starting comprehensive RL training for {num_episodes} episodes...")
        
        all_experiences = []
        training_results = []
        
        for episode in range(num_episodes):
            print(f"🎮 Episode {episode + 1}/{num_episodes}")
            
            # Run episode
            episode_experiences, episode_reward = self.run_rl_training_episode(dataloader)
            
            if episode_experiences:
                all_experiences.extend(episode_experiences)
            
            # Update policy periodically
            if (episode + 1) % self.training_config['update_frequency'] == 0 and len(all_experiences) >= self.rl_policy.ppo_config['batch_size']:
                print(f"   Updating RL policy with {len(all_experiences)} experiences...")
                
                update_results = self.rl_policy.update_policy_ppo(all_experiences)
                
                if update_results:
                    training_results.append({
                        'episode': episode + 1,
                        'update_results': update_results,
                        'total_experiences': len(all_experiences)
                    })
                    
                    print(f"   Policy updated - Mean reward: {update_results.get('mean_reward', 0):.4f}")
                
                # Clear experiences after update
                all_experiences = []
            
            # Evaluation
            if (episode + 1) % self.training_config['evaluation_frequency'] == 0:
                self._evaluate_rl_performance(episode + 1)
        
        print(f"✅ RL training completed: {num_episodes} episodes")
        return training_results
    
    def _evaluate_rl_performance(self, episode):

# ==== 1. FINAL SYSTEM EVALUATOR ====
class FinalSystemEvaluator:
    """Simple but comprehensive system evaluation."""
    
    def __init__(self, model, xai_explainer=None, active_learner=None, rl_policy=None):
        self.model = model
        self.xai_explainer = xai_explainer
        self.active_learner = active_learner
        self.rl_policy = rl_policy
        
        print("📊 Final System Evaluator initialized")
    
    def run_final_evaluation(self, dataloader, max_batches=25):
        """Run complete system evaluation."""
        print("🚀 Running final system evaluation...")
        
        # Metrics storage
        weather_predictions = []
        weather_targets = []
        weather_confidences = []
        inference_times = []
        memory_usage = []
        
        # Component evaluations
        xai_success = 0
        xai_total = 0
        rl_rewards = []
        
        self.model.eval()
        total_samples = 0
        
        with torch.no_grad():
            for batch_idx, batch in enumerate(tqdm(dataloader, desc="Final evaluation")):
                if batch_idx >= max_batches:
                    break
                
                try:
                    # Prepare batch
                    rgb = batch['rgb'].to(device)
                    nir = batch.get('nir')
                    if nir is not None and nir[0] is not None:
                        nir = torch.stack([n if n is not None else torch.zeros_like(rgb[0:1]) for n in nir]).to(device)
                    
                    targets = batch['weather_label'].to(device)
                    total_samples += len(targets)
                    
                    # Model inference with timing
                    start_time = time.time()
                    outputs = self.model(rgb, nir)
                    inference_time = time.time() - start_time
                    
                    # Extract predictions
                    weather_logits = outputs['weather_logits']
                    weather_probs = F.softmax(weather_logits, dim=1)
                    predictions = torch.argmax(weather_logits, dim=1)
                    confidences = torch.max(weather_probs, dim=1)[0]
                    
                    # Store metrics
                    weather_predictions.extend(predictions.cpu().numpy())
                    weather_targets.extend(targets.cpu().numpy())
                    weather_confidences.extend(confidences.cpu().numpy())
                    inference_times.append(inference_time)
                    
                    # Memory tracking
                    if torch.cuda.is_available():
                        memory_usage.append(torch.cuda.memory_allocated() / (1024**3))
                    
                    # Quick XAI test (first few batches)
                    if self.xai_explainer and batch_idx < 3:
                        try:
                            explanation = self.xai_explainer.generate_comprehensive_explanation(rgb[0], targets[0].item())
                            xai_total += 1
                            if explanation.get('gradcam') or explanation.get('attention'):
                                xai_success += 1
                        except:
                            xai_total += 1
                    
                    # Quick RL test (if available)
                    if self.rl_policy and batch_idx < 5:
                        try:
                            system_metrics = {
                                'accuracy_estimate': (predictions == targets).float().mean().item(),
                                'processing_time': inference_time,
                                'memory_usage': memory_usage[-1] if memory_usage else 0.5,
                                'prediction_uncertainty': (1.0 - confidences.mean()).item()
                            }
                            batch_info = {'weather_types': batch['weather_name'], 'difficulty_score': 0.5}
                            
                            state = self.rl_policy.extract_comprehensive_state(outputs, system_metrics, batch_info)
                            action, _, _, _ = self.rl_policy.get_policy_action(state, deterministic=True)
                            reward = self.rl_policy.compute_comprehensive_reward(outputs, action, {'weather_label': targets}, system_metrics)
                            rl_rewards.append(reward.item())
                        except:
                            pass
                
                except Exception as e:
                    print(f"⚠️ Batch {batch_idx} failed: {e}")
                    continue
        
        # Compute final results
        results = self._compute_results(weather_predictions, weather_targets, weather_confidences, 
                                      inference_times, memory_usage, total_samples, 
                                      xai_success, xai_total, rl_rewards)
        
        print("✅ Final evaluation completed")
        return results
    
    def _compute_results(self, predictions, targets, confidences, inf_times, memory_usage, 
                        total_samples, xai_success, xai_total, rl_rewards):
        """Compute comprehensive results."""
        results = {}
        
        # Weather classification
        if predictions and targets:
            predictions = np.array(predictions)
            targets = np.array(targets)
            confidences = np.array(confidences)
            
            accuracy = accuracy_score(targets, predictions)
            f1 = f1_score(targets, predictions, average='weighted', zero_division=0)
            
            # Per-class accuracy
            per_class_acc = {}
            for i, weather_name in enumerate(WEATHER_NAMES):
                class_mask = (targets == i)
                if class_mask.sum() > 0:
                    per_class_acc[weather_name] = float(accuracy_score(targets[class_mask], predictions[class_mask]))
            
            results['weather_classification'] = {
                'accuracy': float(accuracy),
                'f1_score': float(f1),
                'per_class_accuracy': per_class_acc,
                'mean_confidence': float(np.mean(confidences)),
                'total_samples': total_samples
            }
        
        # System performance
        if inf_times:
            times_ms = np.array(inf_times) * 1000
            results['system_performance'] = {
                'mean_inference_time_ms': float(np.mean(times_ms)),
                'fps': float(1.0 / np.mean(inf_times)),
                'p95_inference_time_ms': float(np.percentile(times_ms, 95))
            }
        
        # Memory analysis
        if memory_usage:
            results['memory_analysis'] = {
                'mean_memory_gb': float(np.mean(memory_usage)),
                'peak_memory_gb': float(np.max(memory_usage))
            }
        
        # Component status
        results['components'] = {
            'xai_success_rate': float(xai_success / max(xai_total, 1)),
            'rl_mean_reward': float(np.mean(rl_rewards)) if rl_rewards else 0.0,
            'active_learning_samples': len(self.active_learner.selection_history.get('uncertain_samples', [])) if self.active_learner else 0
        }
        
        # Overall score
        accuracy_score_val = results.get('weather_classification', {}).get('accuracy', 0) * 100
        performance_score = min(results.get('system_performance', {}).get('fps', 0) / 30 * 100, 100)
        overall_score = (accuracy_score_val * 0.6 + performance_score * 0.4)
        
        results['overall'] = {
            'score': float(overall_score),
            'status': '🌟 EXCELLENT' if overall_score >= 85 else '✅ GOOD' if overall_score >= 75 else '👍 READY' if overall_score >= 65 else '⚠️ FAIR'
        }
        
        return results

            print(f"📄 Files: {len(saved_files)} files created")
